' Define record type needed for INTERRUPT
TYPE RegType
	 Ax    AS INTEGER
	 Bx    AS INTEGER
	 Cx    AS INTEGER
	 Dx    AS INTEGER
	 Bp    AS INTEGER
	 Si    AS INTEGER
	 Di    AS INTEGER
	 Flags AS INTEGER
END TYPE

'****************************************************************************
'   Select Office/Test board work station and drive for installation
'============================================================================
SUB GetDiskSpace (SpaceAvail&, Drive$)
'----------------------------------------------------------------------------
	DIM InRegs AS RegType

	InRegs.Ax = &H3600                  ' DOS call for disk free space
	InRegs.Dx = ASC(UCASE$(Drive$)) - 64
	Interrupt &H21, InRegs, InRegs

	IF InRegs.Ax = &HFFFF THEN
		CLS
		COLOR 12, 0
		SpaceAvail& = 0
		Drive$ = "BAD"
	ELSE
		SpaceAvail& = InRegs.Ax
		SpaceAvail& = SpaceAvail& * InRegs.Bx * InRegs.Cx
		Drive$ = ""
	END IF
'****************************************************************************
END SUB
