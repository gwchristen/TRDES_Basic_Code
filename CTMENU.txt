'****************************************************************************
'   Prepares for data edit, calls data entry/edit routine
'============================================================================
SUB EditRecord
'----------------------------------------------------------------------------

	SHARED CallPrg%, EdRec%
	SHARED Value$(), Arow%(), Acol%(), Alen%()
   
	CLS
	RecNum% = OpenFile%
	CLS
	IF RecNum% = 0 THEN
		EXIT SUB
	END IF
	LOCATE 10, 25
	PRINT USING "Record number to edit (1-####)"; RecNum%
	INPUT EdRec%

'   If the user wishes to back out, an entry of zero or
'   more records than there are in the file will do so.

	IF EdRec% < 1 OR EdRec% > RecNum% THEN
		CLOSE
		EXIT SUB
	END IF
	GET #1, EdRec% + 1, Buffer
	Buff$ = Buffer.Buffer
	Parse Buff$
	IF LEFT$(Buff$, 1) <> "1" THEN
		CLOSE
		EXIT SUB
	END IF
	CallPrg% = 0
	Flag% = 4
	CLOSE
	CHAIN "CtData"
'****************************************************************************
END SUB

'****************************************************************************
'   Opens data file.  Returns zero if data file is non-existant or
'   unacceptable, otherwise returns number of records in data file.
'============================================================================
FUNCTION OpenFile%
'----------------------------------------------------------------------------

	SHARED FileName$

OpnFile:
	CLS
	LOCATE 25, 1
	COLOR Mfg%, Mbg%
	PRINT SPACE$(79);
	OPEN FileName$ FOR RANDOM AS #1 LEN = 1000
	GET #1, 1, Info
	IF Info.Number > 0 THEN
		OpenFile% = Info.Number
		EXIT FUNCTION
	END IF
	CLOSE
	KILL FileName$
	CLS
	Cputs "No file found on this disk.", 10
	Cputs "Try again? (Y/N)", 15
	x$ = Get1Key$("YN", "")
	IF x$ = "N" THEN
		OpenFile% = 0
		EXIT FUNCTION
	END IF
	CLS
	x$ = Get1Key$(CHR$(13), "Insert new disk, then press ENTER")
	GOTO OpnFile
'****************************************************************************
END FUNCTION

'****************************************************************************
'   Clears screen, prints menu.  Returns menu choice.
'============================================================================
FUNCTION PrintMenu$
'----------------------------------------------------------------------------
   
	SHARED FileName$

	CLS
	COLOR Mfg%, Mbg%

	InFile% = FREEFILE
	OPEN FileName$ FOR RANDOM AS #InFile% LEN = LEN(Buffer)
	GET #InFile%, 1, Info
	CLOSE #InFile%

'   Clear Status Line

	LOCATE 25, 1, 0
	PRINT SPACE$(79);

'   Locate and draw a box to define menu
	   
	DrawMenuBox 1, 24, 6, 20, 78

'   Print Menu Text

	Cputs Company$, 2
	IF LTRIM$(RTRIM$(Facility$)) > "" THEN
		Cputs Facility$, 3
	END IF
	Cputs "Transformer Record Data Entry System", 4
	Cputs "Current Transformer Menu", 5

'   Menu choice text

	PutXY 8, 24, "1. List Transformer Test Records"
	PutXY 9, 24, "2. List of Transformers in File"
	PutXY 10, 24, "3. Enter Current Transformer data"
	PutXY 11, 24, "4. Edit Transformer Data"
	PutXY 12, 24, "5. Generate CT Records"
	PutXY 16, 24, "C - Change colors"
	PutXY 14, 24, "6. Exit"

	Cputs STR$(Info.Number) + " records in file.", 19

	PutXY 22, 32, "Option number? "

'   Get input
   
	PutXY 2, 77 - LEN(Version$), Version$
   
	COLOR Mfg%, Mbg%
	DO
		DO
			PutXY 5, 3, DATE$
			PutXY 5, 69, TIME$
			PutXY 2, 3, DayOfWeek$(DATE$)
			LOCATE , , 0
			x$ = UCASE$(INKEY$)
		LOOP WHILE x$ = ""
		IF x$ = CHR$(27) THEN
			x$ = "6"
		END IF
	LOOP WHILE INSTR("123456C", x$) = 0

	PrintMenu$ = x$
'****************************************************************************
END FUNCTION
