'============================================================================
'   Dimension array
'----------------------------------------------------------------------------
    DIM SHARED OutFile%, Value$(100), Mds%
    DIM Argv$(20)
'============================================================================
'   Turn on network error trap
'----------------------------------------------------------------------------
    ON ERROR GOTO NetProblem
'============================================================================
    Programmer$ = "K.W.Kroninger"
   
    CLS
   
'   Check command line for Mds1/Mds2 flag

    Cline% = ParseCmdLine%(Argv$())
    FOR Counter% = 1 TO Cline%
        IF INSTR(UCASE$(Argv$(Counter%)), "/MDS2") > 0 THEN
            Mds% = 1
        END IF
        IF InFile$ <> "" AND LEFT$(Argv$(Counter%), 1) <> "/" AND OutFile$ = "" THEN
            OutFile$ = Argv$(Counter%)
        END IF
        IF LEFT$(Argv$(Counter%), 1) <> "/" AND InFile$ = "" THEN
            InFile$ = Argv$(Counter%)
        END IF
    NEXT

    IF InFile$ = "" THEN
        LINE INPUT "Input Filespec:"; InFile$
        InFile$ = LTRIM$(RTRIM$(InFile$))
        IF InFile$ = "" THEN
            END
        END IF
    END IF
 
'   Prompt for output filespec if needed.  Set default to screen.

    IF OutFile$ = "" THEN
        LINE INPUT "Output filespec (just press Enter for screen):"; OutFile$
        OutFile$ = LTRIM$(RTRIM$(OutFile$))
        IF OutFile$ = "" THEN
            OutFile$ = "CON"
        END IF
    END IF
  
'   Open the files

    InFile% = FREEFILE
    OPEN InFile$ FOR INPUT AS #InFile%
    OutFile% = FREEFILE
    OPEN OutFile$ FOR OUTPUT AS #OutFile%
 
'   Here's the loop that does the work.  Read the file, one record, at a time,
'   and call proper subs to process it.  Continue until the end of the input
'   file is reached.

    DO
        x$ = INKEY$                         ' Allow abort with Esc key
        IF x$ = CHR$(27) THEN
            EXIT DO
        END IF
        LINE INPUT #InFile%, Work$
        Code$ = LEFT$(Work$, 1)             ' Get Transaction Code
        SELECT CASE Code$
            CASE "H"
                ParseHeaderRecord (Work$)
                PrintHeaderRecord
            CASE "T"
                ParseTrailerRecord (Work$)
                PrintTrailerRecord
            CASE "1" TO "7"
                ParseDataRecord (Work$)
                PrintDataRecord
            CASE ELSE
                PRINT "ERROR!"
                PRINT Work$
                x$ = Get1Key$(CHR$(13), "Press Enter to continue")
        END SELECT
    LOOP UNTIL EOF(InFile%)
    CLOSE
    CLS

Split:
    END

'============================================================================
'   Error traps
'----------------------------------------------------------------------------
'   Network Problems

NetProblem:
    IF UCASE$(ENVIRON$("DEBUG")) = "YES" THEN
        Junk$ = Get1Key$(CHR$(13) + CHR$(27), STR$(ERR))
    ELSE
        x& = TIMER + 10
        DO
        Junk$ = INKEY$
        LOOP UNTIL Junk$ = CHR$(27) OR x& < TIMER
    END IF
    IF Junk$ = CHR$(27) THEN
        RESUME Split
    ELSE
        RESUME
    END IF

'   End of error traps

'   End of DECODE.BAS main module

'****************************************************************************

'   message if all is not well.

'****************************************************************************
'   Decode and print data record
'============================================================================
SUB PrintDataRecord
'----------------------------------------------------------------------------
    PrintaLine
  
    Prtl "MDS Transaction Code:" + TransactionCode$(Value$(1))
  
    Prtl "Company Code:" + DecodeCompanyCode$(Value$(2))
  
    RequiredData "State Code:", Value$(3)
  
    Prt "Division Code:"
    IF LTRIM$(RTRIM$(Value$(4))) <> "" THEN
        Prtl DecodeDivision$(Value$(4))
    ELSE
        DataMissing
    END IF
  
    RequiredData "Area Code:", Value$(5)
  
    RequiredData "Storeroom Number:", Value$(6)
  
    IF Value$(1) = "5" THEN
        RequiredData "Receiving State Code:", Value$(7)
        Prt "Receiving Division Code:"
        IF LTRIM$(RTRIM$(Value$(8))) <> "" THEN
            Prtl DecodeDivision$(Value$(8))
        ELSE
            DataMissing
        END IF
 
        RequiredData "Receiving Area Code:", Value$(9)
        RequiredData "Receiving Storeroom Code:", Value$(10)
    END IF
  
    IF Value$(1) = "6" OR LTRIM$(RTRIM$(Value$(11))) <> "" THEN
        Prt "Status Code:"
            IF Value$(1) <> "6" OR LTRIM$(RTRIM$(Value$(11))) <> "" THEN
                Prtl Value$(11)
            ELSE
                DataMissing
            END IF
    END IF
  
    IF Value$(1) = "2" OR Value$(1) = "4" THEN
        RequiredData "Purchase Code:", Value$(12)
        Prt "Previous Installed Indicator (Y/N):"
        CheckString "YN", Value$(13)
    END IF
  
    RequiredData "Device Manufacturer Code:", Value$(14)
  
    RequiredData "Device Code:", Value$(15)
  
    RequiredData "Device Serial Number:", Value$(16)
  
    IF Value$(1) = "7" THEN
        RequiredData "Device Code (Accessory Device Only):", Value$(17)
        RequiredData "Accessory Type Code (Accessory Device Only):", Value$(18)
        RequiredData "Manufacturer Code (Accessory Device Only):", Value$(19)
        RequiredData "Battery Check Year (Accessory Device Only):", Value$(20)
    END IF
 
'   If there's no test transaction, this record is done.

    IF Value$(1) > "2" THEN
        EXIT SUB
    END IF
  
    IF LTRIM$(RTRIM$(Value$(21))) <> "" THEN
        Prtl "Associated Manufacturer Code (Register Only):" + Value$(21)
        Prtl "Associated Device Code (Register Only):" + Value$(22)
        Prtl "Associated Device Serial Number (Register Only):" + Value$(23)
    END IF
  
    IF LTRIM$(RTRIM$(Value$(24))) <> "" THEN
        Prtl "Tester Identification:" + Value$(24)
    END IF
  
    Prt "Activity Type Code:"
    IF LTRIM$(RTRIM$(Value$(25))) <> "" THEN
        Prtl ActivityCode$(Value$(25))
    ELSE
        DataMissing
    END IF
  
    Prt "Test Description Code (FD=Field, SH=Shop):"
    IF LTRIM$(RTRIM$(Value$(26))) <> "" THEN
        CheckString "FDSH", Value$(26)
    ELSE
        DataMissing
    END IF
  
    Prt "Test Date (MmDdYyyy):"
    IF LTRIM$(RTRIM$(Value$(27))) <> "" THEN
        Flag% = 0
        FOR c% = 1 TO LEN(Value$(27))
            IF INSTR("0123456789", MID$(Value$(27), c%, 1)) = 0 THEN
                Flag% = 1
            END IF
        NEXT
        IF Flag% = 0 THEN
            Prtl Value$(27)
        ELSE
            Prtl Value$(27) + " - Invalid data in field"
        END IF
    ELSE
        DataMissing
    END IF
    Prtl "Test Occurance Number:" + Value$(28)
    OptionalData "Remarks Regarding Test:", LTRIM$(RTRIM$(Value$(29)))
    IF LTRIM$(RTRIM$(Value$(30))) <> "" THEN
        Prtl "Retirement Reason:" + RetireCode$(Value$(30))
    END IF
  
    Prt "Repair Indicator (Y/N):"
    IF INSTR("YN", Value$(31)) > 0 THEN
        Prtl Value$(31)
    ELSE
        Prtl Value$(31) + " Invalid data in field"
    END IF
    IF LTRIM$(RTRIM$(Value$(32))) <> "" THEN
        Prtl "Damage Code One:" + DamageCode$(Value$(32))
    END IF
    IF LTRIM$(RTRIM$(Value$(33))) <> "" THEN
        Prtl "Damage Code Two:" + DamageCode$(Value$(33))
    END IF
    IF LTRIM$(RTRIM$(Value$(34))) <> "" THEN
        Prtl "Damage Code Three:" + DamageCode$(Value$(34))
    END IF
    IF LTRIM$(RTRIM$(Value$(35))) <> "" THEN

'****************************************************************************
'   Prints data to device/file - no CR/LF
'============================================================================
SUB Prt (Work$)
'----------------------------------------------------------------------------
    PRINT #OutFile%, Work$;
'****************************************************************************
END SUB

'****************************************************************************
'   Prints data to device/file. Appends cr/lf
'============================================================================
SUB Prtl (Work$)
'----------------------------------------------------------------------------
    PRINT #OutFile%, Work$
'****************************************************************************
END SUB

'****************************************************************************
'   Prints identification, then either data or missing message
'============================================================================
SUB RequiredData (Id$, Value$)
'----------------------------------------------------------------------------
    Prt Id$
    IF LTRIM$(RTRIM$(Value$)) <> "" THEN
        Prtl Value$
    ELSE
        DataMissing
    END IF
'****************************************************************************
END SUB

'****************************************************************************
'   Decode retirement code
'============================================================================
FUNCTION RetireCode$ (Pointer$)
'----------------------------------------------------------------------------
    SELECT CASE Pointer$
        CASE "01"
            RetireCode$ = Pointer$ + " - Lost/Stolen"
        CASE "02"
            RetireCode$ = Pointer$ + " - Damaged"
        CASE "03"
            RetireCode$ = Pointer$ + " - Obsolete"
        CASE "04"
            RetireCode$ = Pointer$ + " - Defective"
        CASE "05"
            RetireCode$ = Pointer$ + " - Utility Plant Sold"
        CASE ELSE
            RetireCode$ = Pointer$ + " - Unknown retirement code"
    END SELECT
'****************************************************************************
END FUNCTION

'****************************************************************************
'   Decode MDS transaction code
'============================================================================
FUNCTION TransactionCode$ (Pointer$)
'----------------------------------------------------------------------------
    SELECT CASE Pointer$
        CASE "1"
            TransactionCode$ = Pointer$ + " - Test Record"
        CASE "2"
            TransactionCode$ = Pointer$ + " - Test/History Record"
        CASE "3"
            TransactionCode$ = Pointer$ + " - Physical Inventory Record"
        CASE "4"
            TransactionCode$ = Pointer$ + " - History Record"
        CASE "5"
            TransactionCode$ = Pointer$ + " - Transfer Record"
        CASE "6"
            TransactionCode$ = Pointer$ + " - Status Change Record"
        CASE "7"
            TransactionCode$ = Pointer$ + " - Battery Check Year Record"
        CASE ELSE
            TransactionCode$ = Pointer$ + " - Unknown transaction code"
    END SELECT
'****************************************************************************
END FUNCTION
