DECLARE SUB Prtl (Work$)
'============================================================================
'   Dimension array
'----------------------------------------------------------------------------
    DIM SHARED Outfile%, Value$(100)
    Programmer$ = "K.W.Kroninger"
'============================================================================
'   If input filespec is on command line, use it, otherwise prompt for it.
'----------------------------------------------------------------------------
    CLS
    InFile$ = LTRIM$(RTRIM$(COMMAND$))
    IF InFile$ = "" THEN
        LINE INPUT "Input Filespec:"; InFile$
        InFile$ = LTRIM$(RTRIM$(InFile$))
        IF InFile$ = "" THEN
            END
        END IF
    END IF
  
'   Prompt for output filespec in any case.  Set default to screen.

    LINE INPUT "Output filespec (just press Enter for screen):"; Outfile$
    Outfile$ = LTRIM$(RTRIM$(Outfile$))
    IF Outfile$ = "" THEN
        Outfile$ = "CON"
    END IF
   
'   Open the files

    InFile% = FREEFILE
    OPEN InFile$ FOR INPUT AS #InFile%
    Outfile% = FREEFILE
    OPEN Outfile$ FOR OUTPUT AS #Outfile%
  
'   Here's the loop that does the work.  Read the file, one record, at a time,
'   and call proper subs to process it.  Continue until the end of the input
'   file is reached.

    DO
        x$ = INKEY$                         ' Allow abort with Esc key
        IF x$ = CHR$(27) THEN
            EXIT DO
        END IF
        LINE INPUT #InFile%, Work$
        Code$ = LEFT$(Work$, 1)             ' Get Transaction Code
        SELECT CASE Code$
            CASE "H"
                ParseHeaderRecord (Work$)
                PrintHeaderRecord
            CASE "T"
                ParseTrailerRecord (Work$)
                PrintTrailerRecord
            CASE ELSE
                ParseDataRecord (Work$)
                PrintDataRecord
        END SELECT
    LOOP UNTIL EOF(InFile%)
    CLOSE
    CLS
    END
'****************************************************************************
'   End of INTLIST.BAS main module
'****************************************************************************

'****************************************************************************
'   Parse Elements from Interface File Data Record
'============================================================================

    Value$(45) = MID$(Work$, 249, 4)    ' Seal Code As Left
    Value$(46) = MID$(Work$, 253, 5)    ' Series Percent Accuracy As Found - Full Load
    Value$(47) = MID$(Work$, 258, 5)    ' Series Percent Accuracy As Found - Lag Load

    Value$(50) = MID$(Work$, 273, 5)    ' Series Percent Accuracy As Left - Lag Load
    Value$(51) = MID$(Work$, 278, 5)    ' Series Percent Accuracy As Left - Light Load

    Value$(53) = MID$(Work$, 290, 5)    ' Timing Percent Accuracy As Found
    Value$(54) = MID$(Work$, 295, 5)    ' Timing Percent Accuracy As Left
    Value$(55) = MID$(Work$, 300, 3)    ' Timing Interval Length
    Value$(56) = MID$(Work$, 303, 5)    ' Upscale Percent Accuracy As Found
    Value$(57) = MID$(Work$, 308, 5)    ' Upscale Percent Accuracy As Left

'****************************************************************************
'   Parse Header Record
'============================================================================
SUB ParseHeaderRecord (Work$)
'----------------------------------------------------------------------------
    Work$ = Work$ + SPACE$(50)
    Value$(1) = MID$(Work$, 1, 1)       ' Transaction Code
    Value$(2) = MID$(Work$, 2, 2)       ' Company Code
    Value$(3) = MID$(Work$, 4, 8)       ' Date
'****************************************************************************
END SUB

'****************************************************************************
'   Parse information from trailer record
'============================================================================
SUB ParseTrailerRecord (Work$)
'----------------------------------------------------------------------------
    Work$ = Work$ + SPACE$(50)
    Value$(1) = MID$(Work$, 1, 1)       ' Transaction Code
    Value$(2) = MID$(Work$, 2, 2)       ' Company Code
    Value$(3) = MID$(Work$, 4, 5)       ' Test Record Count
    Value$(4) = MID$(Work$, 9, 5)       ' Test/History Record Count
    Value$(5) = MID$(Work$, 14, 5)      ' Physical Inventory Record Count
    Value$(6) = MID$(Work$, 19, 5)      ' History Record Count
    Value$(7) = MID$(Work$, 24, 5)      ' Transfer Record Count
    Value$(8) = MID$(Work$, 29, 5)      ' Status Change Record Count
    Value$(9) = MID$(Work$, 34, 5)      ' Battery Check Year Record Count
'****************************************************************************
END SUB

'****************************************************************************
'   Prints a line of hyphens
'============================================================================
SUB PrintaLine
'----------------------------------------------------------------------------
    PRINT #Outfile%, STRING$(65, "-")
'****************************************************************************
END SUB
