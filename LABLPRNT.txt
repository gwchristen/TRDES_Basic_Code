'============================================================================
'   Define Record Types
'----------------------------------------------------------------------------
'   Define the type needed for INTERRUPT
'   This one does not handle the segment registers.
    TYPE RegType
        Ax    AS INTEGER
        Bx    AS INTEGER
        Cx    AS INTEGER
        Dx    AS INTEGER
        Bp    AS INTEGER
        Si    AS INTEGER
        Di    AS INTEGER
        Flags AS INTEGER
    END TYPE
'============================================================================
'   Declare procedures
'----------------------------------------------------------------------------
DECLARE SUB Cputs (Message$, Row%)
DECLARE SUB DoVScreen (First%, Pointer%, Title$, Last%, Row%, Col%, Select$(), Sel%)
DECLARE SUB EndOfList (LptFile%)
DECLARE FUNCTION GetFileName$ ()
DECLARE FUNCTION GetVSelection$ (Length%, FileName$, Select$())
DECLARE SUB LoadLookupFile (FileName$, Sel%, Select$(), Title$)
DECLARE SUB PrintALabel (TheList$(), LptFile%, c%)
DECLARE SUB PutXY (Row%, Column%, Message$)
DECLARE SUB ReadTheList (FileName$, TheList$(), Count%)
'============================================================================
    DIM TheList$(100, 2)
    DIM SHARED Select$(100)
'============================================================================

    Programmer$ = "R.W.Oehler/K.W.Kroninger"
   
    CLS

    FileName$ = LTRIM$(RTRIM$(COMMAND$))

    IF FileName$ = "" THEN
        FileName$ = GetFileName$
    END IF

    ReadTheList FileName$, TheList$(), Count%

    Cputs Programmer$, 1

    Cputs "There are" + STR$(Count%) + " records in the file", 2
    
    LptFile% = FREEFILE
   
    OPEN "LPT2:" FOR OUTPUT AS #LptFile%

    FOR c% = 1 TO Count%
        PrintALabel TheList$(), LptFile%, c%
    NEXT
   
    EndOfList LptFile%
   
    END

'****************************************************************************
'   End of main module
'****************************************************************************

'****************************************************************************
'   Centers Message$ on screen line Row%.  Leaves cursor on screen line Row%.
'============================================================================
SUB Cputs (Message$, Row%)
'----------------------------------------------------------------------------
    PutXY Row%, (80 - LEN(Message$)) \ 2, Message$
'****************************************************************************
END SUB

'****************************************************************************
'   Inits Vselect Screen
'============================================================================
SUB DoVScreen (First%, Pointer%, Title$, Last%, Row%, Col%, Select$(), Sel%)
'----------------------------------------------------------------------------
    Pointer% = First%
    CLS
    Cputs Title$, 1
    FOR x% = First% TO Last%
        IF x% = Pointer% THEN
            COLOR 15, 0
        ELSE
            COLOR 7, 0
        END IF
        PutXY Row% + x% - First% + 1, Col%, Select$(x%)
    NEXT
    IF Sel% > 20 THEN
        Cputs "PgUp/PgDn for more selections", 24
    END IF
    Cputs "Up/Down Arrow keys move cursor, Enter Selects, Tab to input", 25
'****************************************************************************
END SUB

'****************************************************************************
'  THE FOLLOWING CODE SPITS OUT 2 LABELS WITH "END OF JOB" ON
'  NORMAL TERMINATION
'============================================================================
SUB EndOfList (LptFile%)
'----------------------------------------------------------------------------

    E$ = CHR$(27)
  
    Q$ = LTRIM$(RTRIM$("L0302" + E$ + "M*END OF JOB"))

    PRINT #LptFile%, E$; "A";
    PRINT #LptFile%, E$; "A3H0820V0001";    ' A3H SEEMS TO SET THE DEFAULT HORZ OFFSET
                                 ' (820 LOOKS GOOD) "V000" SETS VERT. OFFSET

    PRINT #LptFile%, E$; "H15"; E$; "V030"; E$; Q$        ' PRINTS THE LINE OF TEXT
    PRINT #LptFile%, E$; "Q1";         ' Q SEEMS TO BE END OF MESSAGE
    PRINT #LptFile%, E$; "Z";          ' Z SEEMS TO MAKE IT PRINT
    PRINT #LptFile%, E$; "A"; E$; "C"; E$; "Z"

'****************************************************************************
END SUB

'****************************************************************************
'   Gets mail list filename
'============================================================================
FUNCTION GetFileName$
'----------------------------------------------------------------------------
    GetFileName$ = LTRIM$(RTRIM$(GetVSelection$(63, "FILELIST", Select$())))
'****************************************************************************
END FUNCTION

'****************************************************************************
'   Gets one-dimensional selection
'============================================================================
FUNCTION GetVSelection$ (Length%, FileName$, Select$())
'----------------------------------------------------------------------------
  
    DIM InRegs AS RegType, OutRegs AS RegType

    SCREEN , , 1, 1
    CLS
    LoadLookupFile FileName$, Sel%, Select$(), Title$
   
    Row% = 2                ' Init cursor location
    Col% = 4
   
    First% = 1
   
    IF Sel% <= 20 THEN
        Last% = Sel%
    ELSE
        Last% = 20
    END IF
  
    DoVScreen First%, Pointer%, Title$, Last%, Row%, Col%, Select$(), Sel%

    DO
        DO
            x$ = INKEY$
        LOOP WHILE x$ = ""
      
        IF LEN(x$) = 2 THEN
            x$ = RIGHT$(x$, 1)
        END IF
        x% = ASC(x$)
        SELECT CASE x%
            CASE 56, 72                 ' Up arrow or 8
                IF Pointer% > First% THEN
                    COLOR 7, 0
                    PutXY Row% + Pointer% - First% + 1, Col%, Select$(Pointer%)
                    Pointer% = Pointer% - 1
                    COLOR 15, 0
                    PutXY Row% + Pointer% - First% + 1, Col%, Select$(Pointer%)
                END IF
            CASE 50, 80                 ' Down arrow or 2
                IF Pointer% < Last% THEN
                    COLOR 7, 0
                    PutXY Row% + Pointer% - First% + 1, Col%, Select$(Pointer%)
                    Pointer% = Pointer% + 1
                    COLOR 15, 0
                    PutXY Row% + Pointer% - First% + 1, Col%, Select$(Pointer%)
                END IF
            CASE 55, 71                 ' Home key or 7
                IF Pointer% > First% THEN
                    COLOR 7, 0
                    PutXY Row% + Pointer% - First% + 1, Col%, Select$(Pointer%)
                    Pointer% = First%
                    COLOR 15, 0
                    PutXY Row% + Pointer% - First% + 1, Col%, Select$(Pointer%)
                END IF
            CASE 49, 79                 ' End key or 1
                IF Pointer% < Last% THEN
                    COLOR 7, 0
                    PutXY Row% + Pointer% - First% + 1, Col%, Select$(Pointer%)
                    Pointer% = Last%
                    COLOR 15, 0
                    PutXY Row% + Pointer% - First% + 1, Col%, Select$(Pointer%)
                END IF
            CASE 51, 81                 ' Page down key or 3
                IF Sel% > 20 THEN
                    Last% = Last% + 20
                    IF Last% > Sel% THEN
                        Last% = Sel%
                    END IF
                    First% = Last% - 19
                    IF First% < 1 THEN
                        First% = 1
                    END IF
                    DoVScreen First%, Pointer%, Title$, Last%, Row%, Col%, Select$(), Sel%
                END IF
            CASE 57, 73                 ' Page up key or 9
                IF Sel% > 20 THEN
                    First% = First% - 20
                    IF First% < 1 THEN
                        First% = 1
                    END IF
                    Last% = First% + 19
                    IF Last% > Sel% THEN
                        Last% = Sel%
                    END IF
                    DoVScreen First%, Pointer%, Title$, Last%, Row%, Col%, Select$(), Sel%
                END IF
            CASE 9
                CLS
                LOCATE 10, 10
                LINE INPUT "Enter list filename:"; Select$(Pointer%)
                x$ = CHR$(13)
            CASE 13
'                   (Dummy selection to avoid else clause)
            CASE ELSE
                x$ = ""
        END SELECT
    LOOP UNTIL x$ = CHR$(13)
   
    IF INSTR(Select$(Pointer%), "-") THEN
        GetVSelection$ = LEFT$(Select$(Pointer%), INSTR(Select$(Pointer%), "-") - 1)
    ELSE
        GetVSelection$ = Select$(Pointer%)
    END IF
   
    SCREEN , , 0, 0
'****************************************************************************
END FUNCTION

'****************************************************************************
'   Loads a lookup file.  File must be in MRDES II file lookup format.
'   First line is screen title.
'   If a line starts with a semicolon or is blank, it is a comment.
'   Any non blank line that does not start with a semicolon is data.
'============================================================================
SUB LoadLookupFile (FileName$, Sel%, Select$(), Title$)
'----------------------------------------------------------------------------
    Sel% = 0
    Lookup% = FREEFILE
    OPEN FileName$ FOR INPUT AS #Lookup%
    LINE INPUT #Lookup%, Title$
    DO
        LINE INPUT #Lookup%, Work$
        Work$ = LTRIM$(RTRIM$(Work$))
        IF Work$ <> "" AND LEFT$(Work$, 1) <> ";" THEN
            Sel% = Sel% + 1
            Select$(Sel%) = Work$
        END IF
    LOOP UNTIL EOF(Lookup%)
    CLOSE #Lookup%
'****************************************************************************
END SUB

'****************************************************************************
'   Prints a label
'============================================================================
SUB PrintALabel (TheList$(), LptFile%, c%)
'----------------------------------------------------------------------------

    E$ = CHR$(27)
  
    W$ = LTRIM$(RTRIM$("L0202" + E$ + "M"))

    PRINT #LptFile%, E$; "A";
    PRINT #LptFile%, E$; "A3H0820V0001";    ' A3H SEEMS TO SET THE DEFAULT HORZ OFFSET
                              ' (820 LOOKS GOOD) "V000" SETS VERT. OFFSET
                              '  SET THE VERT AND HORZ OFFSET ON EACH LINE!

    PRINT #LptFile%, E$; "H15"; E$; "V001"; E$; W$; TheList$(c%, 0)    ' PRINTS LINE 1 V IS VERT HEIGHT
    PRINT #LptFile%, E$; "H15"; E$; "V040"; E$; W$; TheList$(c%, 1)    ' PRINTS LINE 2
    PRINT #LptFile%, E$; "H15"; E$; "V082"; E$; W$; TheList$(c%, 2)    ' PRINTS LINE 3

    PRINT #LptFile%, E$; "Q1";         ' Q SEEMS TO BE END OF MESSAGE
    PRINT #LptFile%, E$; "Z";          ' Z SEEMS TO MAKE IT PRINT

    PRINT TheList$(c%, 0), TheList$(c%, 1), TheList$(c%, 2)

'****************************************************************************
END SUB

'****************************************************************************
'   Prints string passed in Message$ at screen location Row%, Column%.  Checks
'   Row% and Column% for valid screen coordinates, and Column% plus Message$
'   length the see that the string will fit on line Row%.  Exits without
'   printing string if location invalid.  Leaves cursor on screen line Row%
'============================================================================
SUB PutXY (Row%, Column%, Message$)
'----------------------------------------------------------------------------
    IF Row% < 1 OR Row% > 25 OR Column% < 1 OR Column% > 79 THEN
        EXIT SUB
    END IF
    IF Column% + LEN(Message$) > 80 THEN
        EXIT SUB
    END IF
    LOCATE Row%, Column%
    PRINT Message$;
'****************************************************************************
END SUB

'****************************************************************************
'   This routine will read the names on the list into the TheList$().
'   Data is in the array, Number of labels needed is in Count%
'   This routine written by Ken Kroninger, August, 1996
'============================================================================
SUB ReadTheList (FileName$, TheList$(), Count%)
'----------------------------------------------------------------------------
    InFile% = FREEFILE
    OPEN FileName$ FOR INPUT AS #InFile%
    Count% = 0
    DO
        LINE INPUT #InFile%, Temp$
        Temp$ = LTRIM$(RTRIM$(Temp$))
        IF Temp$ <> "" THEN
            IF INSTR(";!#$%&*'" + CHR$(34), LEFT$(Temp$, 1)) = 0 THEN
                Count% = Count% + 1
                TheList$(Count%, 0) = RTRIM$(LEFT$(Temp$, 21))
                TheList$(Count%, 1) = LTRIM$(RTRIM$(MID$(Temp$, 22, 21)))
                TheList$(Count%, 2) = LTRIM$(RTRIM$(MID$(Temp$, 43, 21)))
            END IF
        END IF
    LOOP UNTIL LEFT$(Temp$, 1) = "#"
    CLOSE #InFile%
'****************************************************************************
END SUB
