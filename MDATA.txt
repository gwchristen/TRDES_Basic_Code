'============================================================================
'   This routine is used to enter/edit Metering Outfit Test record data.
'============================================================================
'   Declare procedures
'----------------------------------------------------------------------------
DECLARE SUB ACode ()
DECLARE SUB Cputs (Message$, Row%)
DECLARE SUB DamageCodes ()
DECLARE SUB DefectCodes ()
DECLARE SUB DisplayKeys (Xfer%, Mfg%, Mbg%, Dfg%, Dbg%)
DECLARE SUB FuncKeys ()
DECLARE FUNCTION Get1Key$ (Accept$, Prompt$)
DECLARE FUNCTION GetVSelection$ (Length%, FileName$, Select$())
DECLARE SUB MoveFieldPointer (Ptr%, Work%(), Vline%())
DECLARE SUB MfrCode ()
DECLARE SUB PrtScrn (Prow%(), Pcol%(), Prompt$(), Arow%(), Acol%(), Value$(), Count%, Company$, Facility$, Mfg%, Mbg%, Dfg%, Dbg%, Number%)
DECLARE SUB Puts (Message$, Row%)
DECLARE SUB PutXY (Row%, Column%, Message$)
DECLARE SUB RetireCodes ()
DECLARE SUB XferData (CallPrg%)
'============================================================================
'   $INCLUDE: 'MDIMCOM.INC'
'============================================================================
	Programmer$ = "K.W.Kroninger"
'   Version check
	IF COMMAND$ <> "" AND LEFT$(UCASE$(LTRIM$(COMMAND$)), 1) = "V" THEN
		PRINT "Metering Outfit/Dual Range CT Data Entry/Edit Utility - Version 9.00"
		END
	END IF
'============================================================================
'   Turn on error handler
'----------------------------------------------------------------------------
	ON ERROR GOTO NetProblem
'============================================================================
'   If returning from MCHECK or MXFER, skip varaible initialization

	IF CallPrg% = 2 OR CallPrg% = 3 THEN
		GOTO SkipInit
	END IF
   
	IF EdRec% > 0 AND CallPrg% = 1 THEN
		CHAIN "MMENU"
	END IF
   
	IF EdRec% > 0 THEN
		GOTO SkipInit
	END IF
   
	IF CallPrg% = 4 THEN
		GOTO PartInit
	END IF

'   Initialize Value Strings

	FOR x% = 1 TO 85
		Value$(x%) = SPACE$(Alen%(x%))
	NEXT

	Value$(1) = LEFT$("M" + SPACE$(Alen%(1)), Alen%(1))
	Value$(7) = RIGHT$(SPACE$(Alen%(7)) + ":1", Alen%(7))
	Value$(8) = RIGHT$(SPACE$(Alen%(8)) + ":1", Alen%(8))
	Value$(17) = LEFT$(DATE$, 2) + MID$(DATE$, 4, 2) + RIGHT$(DATE$, 4)
	Value$(2) = SPACE$(Alen%(2))    ' Purchase Code

PartInit:
	Value$(3) = SPACE$(Alen%(3))    ' Mfg Serial Number
	Value$(12) = SPACE$(Alen%(12))  ' Scrap flag
	Value$(16) = SPACE$(Alen%(16))  '
	Value$(12) = "Y"

	FOR x% = 20 TO 75 STEP 5
		FOR y% = 0 TO 2
			Value$(x% + y%) = SPACE$(Alen%(x% + y%))
		NEXT
	NEXT
	FOR x% = 78 TO 85
		IF x% <> 79 THEN
			IF Arow%(x%) > 0 THEN
				Value$(x%) = SPACE$(Alen%(x%))
			END IF
		END IF
	NEXT

SkipInit:

	PrtScrn Prow%(), Pcol%(), Prompt$(), Arow%(), Acol%(), Value$(), 85, Company$, Facility$, Mfg%, Mbg%, Dfg%, Dbg%, 85
   
	DisplayKeys 1, Mfg%, Mbg%, Dfg%, Dbg%
'============================================================================
'   Input loop

	Ptr% = 1
	x$ = UCASE$(INKEY$)         ' Clear inkey buffer
	LOCATE Arow%(1), Acol%(1) + 1, 1, 0, 7
'============================================================================
   
	Pointer% = 2

WaitForInput:

'   Get the input

	DO
		x$ = UCASE$(INKEY$)
	LOOP WHILE x$ = ""

	IF LEN(x$) > 1 THEN
		SELECT CASE ASC(RIGHT$(x$, 1))
			CASE 59                 ' F1 - Write record to the disk
				Flag% = 0
				CallPrg% = 1
				CHAIN "MCHECK"
			CASE 60                 ' F2 - Return to menu
				GOSUB Escape
			CASE 61                 ' F3 - Data transfer
				COLOR Mfg%, Mbg%
				CallPrg% = 1
				XferData CallPrg%
				GOTO SkipInit
			CASE 63, 80             ' F5 - Move to next data entry field
				GOSUB MoveToNext
			CASE 15, 64, 72         ' F6 - Move to previous data entry field
				GOSUB MoveToPrev
			CASE 65, 77             ' F7 - Move cursor forward
				GOSUB MoveFwd
			CASE 66, 75             ' F8 - Move cursor backward
				GOSUB MoveBwd
			CASE 67, 71             ' F9 - Home cursor
				GOSUB HomeC
			CASE 68                 ' F10 - Help
				COLOR Mfg%, Mbg%
				CLS
				SELECT CASE Ptr%
					CASE 2
						Temp$ = GetVSelection$(Alen%(Ptr%), "ACTIVITY", Select$())
						IF LTRIM$(RTRIM$(Temp$)) <> "" THEN
							Value$(Ptr%) = LEFT$(LTRIM$(RTRIM$(Temp$)) + SPACE$(Alen%(Ptr%)), Alen%(Ptr%))
						END IF
					CASE 4
						Temp$ = GetVSelection$(Alen%(Ptr%), "MFR", Select$())
						IF LTRIM$(RTRIM$(Temp$)) <> "" THEN
							Value$(Ptr%) = LEFT$(LTRIM$(RTRIM$(Temp$)) + SPACE$(Alen%(Ptr%)), Alen%(Ptr%))
						END IF
					CASE 13
						Temp$ = GetVSelection$(Alen%(Ptr%), "REGION", Select$())
						IF LTRIM$(RTRIM$(Temp$)) <> "" THEN
							Value$(Ptr%) = LEFT$(LTRIM$(RTRIM$(Temp$)) + SPACE$(Alen%(Ptr%)), Alen%(Ptr%))
						END IF
					CASE 16
						RetireCodes
					CASE 80 TO 82
						DamageCodes
					CASE 83 TO 85
						DefectCodes
					CASE ELSE
						FuncKeys
				END SELECT
 
				PrtScrn Prow%(), Pcol%(), Prompt$(), Arow%(), Acol%(), Value$(), 85, Company$, Facility$, Mfg%, Mbg%, Dfg%, Dbg%, 85
  
				DisplayKeys 1, Mfg%, Mbg%, Dfg%, Dbg%
  
				LOCATE Arow%(Ptr%), Acol%(Ptr%), 1
				GOTO WaitForInput
			CASE 93                 ' Shift/F10 - Check repair list
				SHELL "RepList"
				COLOR Mfg%, Mbg%
				GOTO SkipInit
		END SELECT
	END IF

'   Perform error check on Esc key
   
	IF x$ = CHR$(27) THEN
		Flag% = 1
		CallPrg% = 1
		CHAIN "MCHECK"
	END IF

'   Enter and Tab = F5
   
	IF x$ = CHR$(9) OR x$ = CHR$(13) THEN
		GOSUB MoveToNext
	END IF

'   BackSpace = F8
   
	IF x$ = CHR$(8) THEN
		GOSUB MoveBwd
	END IF

'   Acceptable input?
   
	IF ASC(x$) < 32 OR ASC(x$) > 90 THEN
		GOTO WaitForInput
	END IF

'   Acceptable Input.  Process it.
   
'   Store the input
   
	MID$(Value$(Ptr%), Pointer%, 1) = x$

'   Point to next character
   
	LOCATE , , 0
	Pointer% = Pointer% + 1

'   Put it on the screen
   
	PRINT x$;

	IF Pointer% <= Alen%(Ptr%) THEN
		LOCATE , , 1, 0, 7
	ELSE
		MoveFieldPointer Ptr%, Fwd%(), Arow%()
		Pointer% = 1
		LOCATE Arow%(Ptr%), Acol%(Ptr%), 1, 0, 7
	END IF
	GOTO WaitForInput
'----------------------------------------------------------------------------
'   End of input loop
'============================================================================
'   Subroutines begin here
'============================================================================
'   Function key 2 - Return to menu
Escape:
	COLOR Mfg%, Mbg%
	LOCATE 25, 1, 0
	x$ = Get1Key$("YN", "Are you sure you want to return to the menu?(Y/N)")
	IF x$ = "Y" THEN
		CallPrg% = 1
		CHAIN "MMENU"
	END IF
   
	PrtScrn Prow%(), Pcol%(), Prompt$(), Arow%(), Acol%(), Value$(), 85, Company$, Facility$, Mfg%, Mbg%, Dfg%, Dbg%, 85
   
	GOTO HomeC
'---------------------------------------------------------------------------
'   Function key 5 - Move to next data entry field
MoveToNext:
	MoveFieldPointer Ptr%, Fwd%(), Arow%()
	GOTO MovePointer
'---------------------------------------------------------------------------
'   Function key 6 - Move to previous data entry field
MoveToPrev:
	MoveFieldPointer Ptr%, Bwd%(), Arow%()
'----------------------------------------------------------------------------
'   The three routines, MoveToNext, MoveToPrev and HomeC, merge here
MovePointer:
	LOCATE , , 0
	SELECT CASE Ptr%
		CASE 2
			IF Value$(Ptr%) = SPACE$(Alen%(Ptr%)) THEN
				Temp$ = GetVSelection$(Alen%(Ptr%), "ACTIVITY", Select$())
				IF LTRIM$(RTRIM$(Temp$)) <> "" THEN
					Value$(Ptr%) = LEFT$(LTRIM$(RTRIM$(Temp$)) + SPACE$(Alen%(Ptr%)), Alen%(Ptr%))
				END IF
			END IF
		CASE 4
			IF Value$(Ptr%) = SPACE$(Alen%(Ptr%)) THEN
				Temp$ = GetVSelection$(Alen%(Ptr%), "MFR", Select$())
				IF LTRIM$(RTRIM$(Temp$)) <> "" THEN
					Value$(Ptr%) = LEFT$(LTRIM$(RTRIM$(Temp$)) + SPACE$(Alen%(Ptr%)), Alen%(Ptr%))
				END IF
			END IF
		CASE 13
			IF Value$(Ptr%) = SPACE$(Alen%(Ptr%)) THEN
				Temp$ = GetVSelection$(Alen%(Ptr%), "REGION", Select$())
				IF LTRIM$(RTRIM$(Temp$)) <> "" THEN
					Value$(Ptr%) = LEFT$(LTRIM$(RTRIM$(Temp$)) + SPACE$(Alen%(Ptr%)), Alen%(Ptr%))
				END IF
			END IF
	END SELECT
	COLOR Dfg%, Dbg%
	PutXY Arow%(Ptr%), Acol%(Ptr%), Value$(Ptr%)
	LOCATE Arow%(Ptr%), Acol%(Ptr%), 1, 0, 7
	Pointer% = 1
	RETURN WaitForInput
'---------------------------------------------------------------------------
'   Function key 7 - Move cursor forward
MoveFwd:
	LOCATE , , 0
	Pointer% = Pointer% + 1
	IF Pointer% <= Alen%(Ptr%) THEN
		LOCATE , POS(0) + 1, 1, 0, 7
	ELSE
		MoveFieldPointer Ptr%, Fwd%(), Arow%()
		Pointer% = 1
		LOCATE Arow%(Ptr%), Acol%(Ptr%), 1, 0, 7
	END IF
	RETURN WaitForInput
'---------------------------------------------------------------------------
'   Function key 8 - Move cursor backward
MoveBwd:
	Pointer% = Pointer% - 1
	LOCATE , , 0
	IF Pointer% > 0 THEN
		LOCATE Arow%(Ptr%), POS(0) - 1, 1, 0, 7
		RETURN WaitForInput
	ELSE
		MoveFieldPointer Ptr%, Bwd%(), Arow%()
		LOCATE Arow%(Ptr%), Acol%(Ptr%) + Alen%(Ptr%) - 1, 1, 0, 7
		Pointer% = Alen%(Ptr%)
	END IF
	RETURN WaitForInput
'---------------------------------------------------------------------------
'   F9 - Home cursor
HomeC:
	Ptr% = 1
	Pointer% = 1
	GOTO MovePointer
'============================================================================
'   Error traps
'----------------------------------------------------------------------------
'   Network Problems

NetProblem:
	IF UCASE$(ENVIRON$("DEBUG")) = "YES" THEN
		Junk$ = Get1Key$(CHR$(13) + CHR$(27), STR$(ERR))
	ELSE
		x& = TIMER + 10
		SELECT CASE ERR
			CASE 57, 71
				Junk$ = "Drive not available"
			CASE ELSE
				Junk$ = "Error" + STR$(ERR)
		END SELECT
		Cputs Junk$ + " - Enter to retry, Esc to exit", 25
		DO
		Junk$ = INKEY$
		LOOP UNTIL Junk$ = CHR$(13) OR Junk$ = CHR$(27) OR x& < TIMER
		Cputs SPACE$(78), 25
	END IF
	IF Junk$ = CHR$(27) THEN
		RESUME WaitForInput
	ELSE
		RESUME
	END IF

'   End of error traps
'****************************************************************************
'   End of MDATA main module
'****************************************************************************

'****************************************************************************
'   Transfer data from Knopp test set.
'============================================================================
SUB XferData (CallPrg%)
'----------------------------------------------------------------------------
	SHARED Value$(), Alen%()

XMenu:
	CLS
	COLOR Mfg%, Mbg%
	LOCATE 1, 20
	Cputs CHR$(201) + STRING$(58, 205) + CHR$(187), 1
	FOR x% = 1 TO 4
		Cputs CHR$(186) + STRING$(58, 32) + CHR$(186), CSRLIN + 1
	NEXT
	Cputs CHR$(200) + STRING$(58, 205) + CHR$(188), CSRLIN + 1
	Cputs Company$, 2
	Cputs Facility$, 3
	Cputs "Transformer Record Data Entry System", 4
	Cputs "Knopp/TRDES Data Transfer", 5
	PutXY 7, 1, "# Burden Amps Ratio  Phase  Accy     # Burden Amps Ratio  Phase  Accy"
	LOCATE 8, 22
	PRINT "Error  Class                         Error  Class"
	FOR x% = 1 TO 6
		r% = (x% - 1) * 5 + 18
		s% = r% + 30
		x$ = CHR$(x% + 64)
		r$ = CHR$(x% + 70)
		PRINT x$; "  "; Value$(r% + 1); "  "; Value$(r%); " "; Value$(r% + 2); " ";
		PRINT Value$(r% + 3); " "; Value$(r% + 4); "   ";
		PRINT r$;
		PRINT "  "; Value$(s% + 1); "  "; Value$(s%); " "; Value$(s% + 2); " ";
		PRINT Value$(s% + 3); " "; Value$(s% + 4)
	NEXT
	PRINT
	PRINT TAB(20); "7. Return to data entry screen"
	PRINT
	PRINT
	PRINT TAB(20); "MFG S/N  "; Value$(3)
	PRINT TAB(20); "MFG:     "; Value$(4)
	PRINT TAB(20); "TYPE:    "; Value$(5)
	PRINT
	PRINT TAB(13); "Please select A-L for line number to transfer data to,"
	PRINT TAB(13); "or 7 to return to data entry screen.";
	x$ = Get1Key$("ABCDEFGHIJKL7", "")
	x% = INSTR("ABCDEFGHIJKL7", x$)
	IF x% = 13 THEN
		CallPrg% = 3
		EXIT SUB
	END IF
	r% = (x% - 1) * 5 + 18
	Cb% = FREEFILE
	OPEN "COM1:1200,N,8,1,DS,CS" FOR INPUT AS #Cb% LEN = 500
	CLS
	LOCATE 10, 16
	PRINT "Press HOLD/PRINT key when ready to transfer data."

'   Wait for first character

	WHILE LOC(Cb%) = 0
		PRINT ".";
	WEND

'   First character received, wait 10 seconds, then input what's in the buffer

	CLS
	PutXY 10, 33, "Receiving Data"
	x& = TIMER + 10
	WHILE TIMER < x&
	WEND
	a$ = INPUT$(LOC(Cb%), #Cb%)

	CLOSE #Cb%
	CLS
	PRINT a$
	CLS
	PutXY 10, 34, "Parsing data"

'   Milliradians or minutes?

	MilliRad% = ABS(INSTR(a$, "MILLIRAD") > 0)

'   Ratio correction factor, or error?

	Factor% = ABS(INSTR(a$, "CORR") = 0)

'   Get rid of header information we don't need

	a% = INSTR(a$, "(%)")
	a$ = MID$(a$, a% + 3)
	a% = INSTR(a$, "(%)")
	IF a% > 0 THEN
		a$ = MID$(a$, a% + 3)
	END IF

'   Now move over to data

	WHILE INSTR("0123456789.+-)", LEFT$(a$, 1)) = 0
		a$ = MID$(a$, 2)
	WEND

'   Grab ampere reading

	Amp$ = ""
	WHILE LEFT$(a$, 1) <> " "
		Amp$ = Amp$ + LEFT$(a$, 1)
		a$ = MID$(a$, 2)
	WEND

'   Remove leading spaces

	a$ = LTRIM$(a$)

'   Ratio error or correction factor
'   Correction in Ratio$= line?

	Ratio$ = ""
	WHILE LEFT$(a$, 1) <> " "
		Ratio$ = Ratio$ + LEFT$(a$, 1)
		a$ = MID$(a$, 2)
	WEND

'   If it's Ratio Error, change it to Ratio

	IF Factor% <> 0 THEN
		Corr = VAL(Ratio$) / 100
		Ratio = (Corr + 1) * 100
	ELSE
		Ratio = VAL(Ratio$) * 100
	END IF
   
	Ratio$ = STR$((INT(Ratio * 100 + .5)) / 100)
  
	IF INSTR("0123456789", LEFT$(Ratio$, 1)) = 0 THEN
		Ratio$ = MID$(Ratio$, 2)
	END IF

'   More spaces

	a$ = LTRIM$(a$)

'   Now grab the phase error

	Phase$ = ""
	WHILE LEFT$(a$, 1) <> " "
		Phase$ = Phase$ + LEFT$(a$, 1)
		a$ = MID$(a$, 2)
	WEND

'   Convert milliradians to minutes, if necessary

	IF MilliRad% <> 0 THEN
		Radian = VAL(Phase$)
		Phase = Radian * 3.43774678#
	ELSE
		Phase = VAL(Phase$)
	END IF
	Phase$ = STR$((INT(Phase * 100 + .5)) / 100)
  
	IF LEFT$(Phase$, 1) = " " THEN
		MID$(Phase$, 1, 1) = "+"
	END IF

'   More spaces

	a$ = LTRIM$(a$)

'   Accuracy class, or some fantasy thereof

	AccClass$ = ""
	WHILE LEFT$(a$, 1) <> " " AND LEFT$(a$, 1) <> CHR$(13) AND LEN(a$) > 1
		AccClass$ = AccClass$ + LEFT$(a$, 1)
		a$ = MID$(a$, 2)
	WEND

'   Now put the results where they belong.

	Value$(r%) = LEFT$(Amp$ + SPACE$(Alen%(r%)), Alen%(r%))
	Value$(r% + 2) = LEFT$(Ratio$ + SPACE$(Alen%(r% + 2)), Alen%(r% + 2))
	Value$(r% + 3) = LEFT$(Phase$ + SPACE$(Alen%(r% + 3)), Alen%(r% + 3))
	Value$(r% + 4) = LEFT$(AccClass$ + SPACE$(Alen%(r% + 4)), Alen%(r% + 4))

'   And go back to menu

	GOTO XMenu
'****************************************************************************
END SUB
