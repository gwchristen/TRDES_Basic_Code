	END IF

'****************************************************************************
'   Sends TRDES barcode data to office computer.
'============================================================================
SUB Barcode (LocalFile$, BcodeFile$, SerialFile$)
'----------------------------------------------------------------------------

	DIM Info AS InfoRecord, DataRec AS DataRecord

	DrawMenuBox 9, 11, 0, 0, 40
  
	Cputs "You are sending barcode records (Y/N)?", 10
  
	x$ = Get1Key$("YN", "")
  
	IF x$ = "N" THEN
		EXIT SUB
	END IF
  
	CLS
   
	LocalFile% = FREEFILE
   
	OPEN LocalFile$ FOR RANDOM AS #LocalFile% LEN = LEN(DataRec)

	GET #LocalFile%, 1, Info

	IF Info.Number = 0 THEN
		CLOSE #LocalFile%
		Cputs "No data to send", 10
		Cputs "Press Enter to exit", 15
		x$ = Get1Key$(CHR$(13), "")
		EXIT SUB
	END IF

'   Delete any remote serial number file

	Serials " ", 1, SerialFile$

'   If we get here, there's records to process

'   Send all, or part of file?

	DrawMenuBox 9, 11, 0, 0, 40
   
	Cputs "All, or Part of file?", 10

	x$ = Get1Key$("AP", "(A/P)")

	IF x$ = "A" THEN
		First% = 1
		Last% = Info.Number
	ELSE
		CLS
		LOCATE 8, 15
		INPUT "Starting record to send: "; Start%
		IF Start% < 1 THEN
			Start% = 1
		END IF
		IF Start% > Info.Number THEN
			CLOSE
			EXIT SUB
		END IF
		LOCATE 14, 15
		INPUT "Ending record to send: "; Finish%
		IF Finish% < Start% THEN
			CLOSE
			EXIT SUB
		END IF
		IF Finish% > Info.Number THEN
			Finish% = Info.Number
		END IF
		First% = Start%
		Last% = Finish%

	END IF

	FOR Counter% = First% TO Last%
	   
		GET #1, Counter% + 1, DataRec

		Typ$ = LEFT$(DataRec.Buffer, 1)

		SELECT CASE Typ$
			CASE "1"
				Ct DataRec.Buffer, SerialFile$, BcodeFile$
			CASE "3"
				Vt DataRec.Buffer, SerialFile$, BcodeFile$
			CASE "6"
				Rt DataRec.Buffer, SerialFile$, BcodeFile$
			CASE "7"
				MtrOutFit DataRec.Buffer, SerialFile$, BcodeFile$
		END SELECT

	NEXT

	CLOSE #LocalFile%

	SHELL "cleanup"
'****************************************************************************
END SUB

'****************************************************************************
'   Process CT record
'============================================================================
SUB Ct (Buffer$, SerialFile$, BcodeFile$)
'----------------------------------------------------------------------------

'   Code - This will always be 5 characters.
  
	Code$ = MID$(Buffer$, 2, 5)

'   MFR Serial Number - Will always be non-blank, but can be any length from
'   1 to 26 characters.  We print exactly 9, prefixing zeros as necessary.
		 
	Serial$ = MID$(Buffer$, 453, 26)
	Serial$ = LTRIM$(RTRIM$(Serial$))
 
'   Save serial number for label print

	Serials Serial$, 2, SerialFile$

'   Format serial number for barcode print

	Serial$ = RIGHT$("00000000000" + Serial$, 11)

'   Mfr - TRDES will use MRS/MDS mfr code, so look up BC Mfr code.
		 
	Junk$ = MID$(Buffer$, 28, 2)
	Mfr$ = MfrLookup$(Junk$)
  
'   Call the write module

	WriteRecord Code$, Serial$, Mfr$, BcodeFile$
'****************************************************************************
END SUB

'****************************************************************************
'   Looks up bar code Mfr code from MRS/MDS code.
'   Nothing special here, it's really a simple lookup table.
'============================================================================
FUNCTION MfrLookup$ (Work$)
'----------------------------------------------------------------------------
	SELECT CASE Work$
		CASE "04"               ' General Electric
			Mfr$ = "D"
		CASE "06"               ' Schlumberger
			Mfr$ = "F"
		CASE "07"               ' ABB
			Mfr$ = "G"
		CASE "09"               ' Superior Technology
			Mfr$ = "I"
		CASE "11"               ' Associated Engineering
			Mfr$ = "K"
		CASE "12"               ' Ferranti-Packard
			Mfr$ = "L"
		CASE "14"               ' Astra
			Mfr$ = "N"
		CASE "16"               ' Meter Devices
			Mfr$ = "P"
		CASE "26"               ' Electromagnetic Ind/Square D
			Mfr$ = "Z"
		CASE ELSE
			Mfr$ = "0"
	END SELECT
 
'   Return the value

	MfrLookup$ = Mfr$
'****************************************************************************
END FUNCTION

'****************************************************************************
'   Process Metering Outfit Record
'============================================================================
SUB MtrOutFit (Buffer$, SerialFile$, BcodeFile$)
'----------------------------------------------------------------------------

'   Code - This will always be 5 characters.

	Code$ = MID$(Buffer$, 2, 5)

'   MFR Serial Number - Will always be non-blank, but can be any length from
'   1 to 26 characters.  We print exactly 9, prefixing zeros as necessary.

	Serial$ = LTRIM$(RTRIM$(MID$(Buffer$, 485, 26)))
 
'   Save serial number for label print

	Serials Serial$, 2, SerialFile$

'   Format serial number for barcode print

	Serial$ = RIGHT$("00000000000" + Serial$, 11)

'   Mfr - TRDES will use MRS/MDS mfr code, so look up BC Mfr code.

	Junk$ = MID$(Buffer$, 36, 2)
	Mfr$ = MfrLookup$(Junk$)
  
	WriteRecord Code$, Serial$, Mfr$, BcodeFile$

'****************************************************************************
END SUB

'   Locate and draw a box to define menu

	IF Facility$ > "" THEN

	END IF

	Cputs "Main Menu", 5

	PutXY 11, 28, "3. Reactiformers"
	PutXY 12, 28, "4. Metering Outfits"

	PutXY 18, 28, "S - Send complete data to MDS"
	PutXY 19, 28, "B - Send barcode information to MDS"

'****************************************************************************
'   Read menu.ini file.  File is a somewhat freeform text file.  The only
'   line actually required for this procedure to function is the line
'   beginning with an octothorpe, which is used to signal the end of the
'   file.  For proper program operation, the lines beginning with local,
'   remote, barcode and serial must exist, and point to the proper
'   path/filenames for operation.
'============================================================================
SUB ReadIniFile (LocalFile$, Remote$, BcodeFile$, SerialFile$, Facility$)
'----------------------------------------------------------------------------
	CLS
	IniFile% = FREEFILE
	OPEN "menu.ini" FOR INPUT AS #IniFile%
	DO
		LINE INPUT #IniFile%, Work$
		PRINT Work$
		Work$ = LTRIM$(RTRIM$(Work$))
		IF INSTR("#;", LEFT$(Work$, 1)) = 0 AND Work$ <> "" THEN
			IF UCASE$(LEFT$(Work$, 5)) = "LOCAL" THEN
				LocalFile$ = LTRIM$(MID$(Work$, INSTR(Work$, "=") + 1))
				IF INSTR(LocalFile$, " ") THEN
					LocalFile$ = LEFT$(LocalFile$, INSTR(LocalFile$, " ") - 1)
				END IF
			END IF
			IF UCASE$(LEFT$(Work$, 6)) = "REMOTE" THEN
				Remote$ = LTRIM$(MID$(Work$, INSTR(Work$, "=") + 1))
				IF INSTR(Remote$, " ") THEN
					Remote$ = LEFT$(Remote$, INSTR(LocalFile$, " ") - 1)
				END IF
			END IF
			IF UCASE$(LEFT$(Work$, 7)) = "BARCODE" THEN
				BcodeFile$ = LTRIM$(MID$(Work$, INSTR(Work$, "=") + 1))
				IF INSTR(BcodeFile$, " ") THEN
					Bcode$ = LEFT$(BcodeFile$, INSTR(BcodeFile$, " ") - 1)
				END IF
			END IF
			IF UCASE$(LEFT$(Work$, 6)) = "SERIAL" THEN
				SerialFile$ = LTRIM$(MID$(Work$, INSTR(Work$, "=") + 1))
				IF INSTR(SerialFile$, " ") THEN
					SerialFile$ = LEFT$(SerialFile$, INSTR(Serial$, " ") - 1)
				END IF
			END IF
			IF UCASE$(LEFT$(Work$, 8)) = "FACILITY" THEN
				Facility$ = LTRIM$(RTRIM$(MID$(Work$, INSTR(Work$, "=") + 1)))
			END IF
		END IF
	LOOP UNTIL LEFT$(Work$, 1) = "#"
	CLOSE #IniFile%
'****************************************************************************
END SUB

'****************************************************************************
'   Process Reactiformer record
'============================================================================
SUB Rt (Buffer$, SerialFile$, BcodeFile$)
'----------------------------------------------------------------------------
  
	Code$ = MID$(Buffer$, 2, 5)

'   MFR Serial Number - Will always be non-blank, but can be any length from
'   1 to 26 characters.  We print exactly 9, prefixing zeros as necessary.

	Serial$ = LTRIM$(RTRIM$(MID$(Buffer$, 301, 26)))
 
'   Save serial number for label print
  
	Serials Serial$, 2, SerialFile$

'   Format serial number for barcode label print

	Serial$ = RIGHT$("00000000000" + Serial$, 11)

'   Mfr - TRDES will use MRS/MDS mfr code, so look up BC Mfr code.
'   Code - This will always be 5 characters.
  
	Junk$ = MID$(Buffer$, 28, 2)
	Mfr$ = MfrLookup$(Junk$)
  
	WriteRecord Code$, Serial$, Mfr$, BcodeFile$

'****************************************************************************
END SUB

'****************************************************************************
'   Copy TRDES records to hard drive area where GATHER can pick them up.
'============================================================================
SUB SendRecords (LocalFile$, Remote$)
'----------------------------------------------------------------------------
   
	DIM SrcInfo AS InfoRecord, DstInfo AS InfoRecord, DataRec AS DataRecord

	DrawMenuBox 9, 11, 0, 0, 41
   
	Cputs "You are sending complete records (Y/N)?", 10
   
	x$ = Get1Key$("YN", "")
   
	IF x$ = "N" THEN
		EXIT SUB
	END IF
   
   
	DrawMenuBox 9, 11, 0, 0, 42

	Cputs "(A)ll records, or (P)art of the records?", 10
   
	x$ = Get1Key$("AP" + CHR$(27), "A, P, or Esc to return to main menu")

	DrawMenuBox 9, 11, 0, 0, 40
   
	LocalFile% = FREEFILE
   
	OPEN LocalFile$ FOR RANDOM AS #LocalFile% LEN = LEN(DataRec)

	GET #LocalFile%, 1, SrcInfo
   
	IF SrcInfo.Number = 0 THEN
		CLOSE #LocalFile%
		Cputs "No data to send", 10
		Cputs "Press Enter to exit", 15
		x$ = Get1Key$(CHR$(13), "")
		EXIT SUB
	END IF

	RemoteFile% = FREEFILE

	OPEN Remote$ FOR RANDOM AS #RemoteFile% LEN = LEN(DataRec)

	GET #RemoteFile%, 1, DstInfo

	IF DstInfo.Number = 0 THEN
			DstInfo.Signature = "TRDES"
	END IF
   
	IF x$ = "A" THEN
		Start% = 1
		Finish% = SrcInfo.Number
	ELSE
		CLS
		LOCATE 8, 15
		INPUT "Starting record to send: "; Start%
		IF Start% < 1 THEN
			Start% = 1
		END IF
		IF Start% > SrcInfo.Number THEN
			CLOSE
			EXIT SUB
		END IF
		LOCATE 14, 15
		INPUT "Ending record to send: "; Finish%
		IF Finish% < Start% THEN
			CLOSE
			EXIT SUB
		END IF
		IF Finish% > SrcInfo.Number THEN
			Finish% = SrcInfo.Number
		END IF
	END IF
   
	FOR Counter% = Start% TO Finish%
	  
'       Add a note to let the operator know something's happening

		Cputs "Processing record" + STR$(Counter%) + " of" + STR$(SrcInfo.Number), 10
		GET #LocalFile%, Counter% + 1, DataRec          ' Get the record
		DstInfo.Number = DstInfo.Number + 1             ' Increment the count
		PUT #RemoteFile%, DstInfo.Number + 1, DataRec   ' Put the record
	NEXT

	CLOSE #LocalFile%

	PUT #RemoteFile%, 1, DstInfo

	CLOSE #RemoteFile%
		   
	IF x$ = "A" THEN
		x% = DateBack%("TRSDAT.DAT")
	ELSE
		SHELL "copy " + LocalFile$ + " trsdat"
		x% = DateBack%("TRSDAT")
		LocalFile% = FREEFILE
		OPEN LocalFile$ FOR RANDOM AS #LocalFile% LEN = LEN(DataRec)
		RemoteFile% = FREEFILE
		OPEN "Work" FOR RANDOM AS #RemoteFile% LEN = LEN(DataRec)
		GET #LocalFile%, 1, SrcInfo
		DstInfo.Number = 0
		DstInfo.Signature = "TRDES"
		FOR Counter% = 1 TO SrcInfo.Number
			IF Counter% < Start% OR Counter% > Finish% THEN
				GET #LocalFile%, Counter% + 1, DataRec
				DstInfo.Number = DstInfo.Number + 1
				PUT #RemoteFile%, DstInfo.Number + 1, DataRec
			END IF
		NEXT
		CLOSE #LocalFile%
		PUT #RemoteFile%, 1, DstInfo
		CLOSE #RemoteFile%
		SHELL "copy Work " + LocalFile$
		x% = DateBack%("Work")
	END IF

'****************************************************************************
END SUB

'****************************************************************************
'   Saves transformer serial number to file for later label print.  Call with
'   Func%=1 to initialize file, Func%=2 to append data to file.
'============================================================================
SUB Serials (Serial$, Func%, FileName$)
'----------------------------------------------------------------------------

	So% = FREEFILE
	IF Func% = 1 THEN
		OPEN FileName$ FOR OUTPUT AS #So%
	ELSE
		OPEN FileName$ FOR APPEND AS #So%
		PRINT #So%, Serial$
	END IF
	CLOSE #So%

'****************************************************************************
END SUB

'****************************************************************************
'   Assemble and write the record
'============================================================================
SUB WriteRecord (c$, s$, m$, FileName$)
'----------------------------------------------------------------------------
	Work$ = m$ + s$ + c$
	Lbl% = FREEFILE
	OPEN FileName$ FOR APPEND AS #Lbl%
	PRINT Work$
	PRINT #Lbl%, Work$
	CLOSE #Lbl%
'****************************************************************************
END SUB
