		IF x$ <> CHR$(27) THEN
			GET #1, Ptr% + 1, Buffer
			Buff$ = Buffer.Buffer
			x$ = LEFT$(Buff$, 1)
			IF x$ = "7" THEN
				Parse Buff$
				LineCount% = PrintLine%(LineCount%, Ptr%, Value$())
			END IF
			IF LineCount% >= 20 THEN
				x$ = Get1Key$(CHR$(13) + CHR$(27), "ENTER to continue, ESC to return to menu")
				IF x$ = CHR$(13) THEN
					LineCount% = Heading%
				END IF
			END IF
		END IF
	NEXT
	CLOSE
	IF LineCount% > 0 THEN
		LOCATE 25, 1
		PRINT SPACE$(79);
		LOCATE 25, 23
		PRINT "Press ENTER to return to the menu";
		DO
			x$ = INPUT$(1)
		LOOP WHILE x$ <> CHR$(13)
	END IF

		CLOSE
		KILL FileName$
		CLS
		PutXY 10, 20, "No file found on this disk."
		PutXY 15, 20, "Try again? (Y/N)"
		x$ = Get1Key$("YN", "")
		IF x$ = "N" THEN
			RETURN
		END IF
		CLS
		x$ = Get1Key$(CHR$(13), "Insert new disk, then press ENTER")
		GOTO OpnFile
	END IF

	RETURN
'============================================================================
'   Subroutine to edit record

EditRecord:
	CLS
	GOSUB OpnFile
	CLS
	IF RecNum% = 0 THEN
		GOTO Menu
	END IF
	LOCATE
	PRINT USING "Record number to edit (1-####)"; RecNum%

'   If the user wishes to back out, an entry of zero or
'   more records than there are in the file will do so.

	IF EdRec% < 1 OR EdRec% > RecNum% THEN
		CLOSE
		GOTO Menu
	END IF
   
	GET #1, EdRec% + 1, Buffer
   
	Buff$ = Buffer.Buffer
	x$ = LEFT$(Buff$, 1)
	IF x$ <> "7" THEN
		CLOSE
		GOTO Menu
	END IF
	Parse Buff$
	CallPrg% = 0
	Flag% = 4
	CLOSE
	CHAIN "MDATA"

	PutXY 15, 24, "Please correct it and press ENTER"

	RESUME OpenIt
'----------------------------------------------------------------------------
'   Network Problems

NetProblem:
	IF UCASE$(ENVIRON$("DEBUG")) = "YES" THEN
		Junk$ = Get1Key$(CHR$(13) + CHR$(27), STR$(ERR))
	ELSE
		x& = TIMER + 10
		SELECT CASE ERR
			CASE 57, 71
				Junk$ = "Drive not available"
			CASE ELSE
				Junk$ = "Error" + STR$(ERR)
		END SELECT
		Cputs Junk$ + " - Enter to retry, Esc to exit", 25
		DO
		Junk$ = INKEY$
		LOOP UNTIL Junk$ = CHR$(13) OR Junk$ = CHR$(27) OR x& < TIMER
		Cputs SPACE$(78), 25
	END IF
	IF Junk$ = CHR$(27) THEN
		RESUME Menu
	ELSE
		RESUME
	END IF

'   End of error traps

	SHARED FileName$

	CLS
   
	COLOR Mfg%, Mbg%

	InFile% = FREEFILE
	OPEN FileName$ FOR RANDOM AS #InFile% LEN = LEN(Buffer)
	GET #InFile%, 1, Info
	CLOSE #InFile%

'   Clear Status Line

	PutXY 25, 1, SPACE$(79)

'   Locate and draw a box to define menu
	  
	DrawMenuBox 1, 24, 6, 20, 78

'   Print Menu Text

	Cputs Company$, 2
	Cputs Facility$, 3
	Cputs "Transformer Record Data Entry System", 4
	Cputs "Metering Outfit Menu", 5

'   Menu choice text

	COLOR Mfg%, Mbg%
	PutXY 8, 22, "1. List Metering Outfit Test Records"
	PutXY 9, 22, "2. List of Metering Outfits in File"
	PutXY 10, 22, "3. Enter Metering Outfit Data"
	PutXY 11, 22, "4. Edit Metering Outfit Data"
	PutXY 12, 22, "5. Generate Metering Outfit Records"
	PutXY 14, 22, "6. Exit"
	PutXY 16, 24, "C - Change colors"
	   
	Cputs STR$(Info.Number%) + " records in file.", 19

	PutXY 22, 32, "Option number? "

'   Get input
  
	PutXY 2, 77 - LEN(Version$), Version$
	DO
		DO
			PutXY 5, 3, DATE$
			PutXY 5, 69, TIME$
			PutXY 2, 3, DayOfWeek$(DATE$)
			LOCATE , , 0
			x$ = UCASE$(INKEY$)
		LOOP WHILE x$ = ""
		IF x$ = CHR$(27) THEN
			x$ = "6"
		END IF
	LOOP WHILE INSTR("123456C", x$) = 0

	PrintMenu$ = x$
'****************************************************************************
END FUNCTION
