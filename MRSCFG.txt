'   July, 1994

'============================================================================
'   Declare procedures
'----------------------------------------------------------------------------
DECLARE SUB Cputs (Message$, Row%)
DECLARE SUB DrawMenuBox (First%, Last%, a%, b%, LengTh%)
DECLARE FUNCTION Get1Key$ (Accept$, Prompt$)
DECLARE SUB PutXY (Row%, Column%, Message$)
DECLARE SUB ToggleValue (Value%, Max%)
'============================================================================
	Programmer$ = "K.W.Kroninger"

'   Check for version number

	IF UCASE$(LTRIM$(RTRIM$(COMMAND$))) = "V" THEN
		PRINT "MRDES II Configuration Utility - Version 2.00"
		END
	END IF
'----------------------------------------------------------------------------
'   Dimension the array.

	DIM Value%(10)
'----------------------------------------------------------------------------
'   Input the menu titles

	Buffer% = FREEFILE
	OPEN "DATAFILE" FOR INPUT AS #Buffer%
	LINE INPUT #Buffer%, Heading$
	LINE INPUT #Buffer%, Facility$
	CLOSE

'   Read in the configuration data file, if it exists.

	ON ERROR GOTO NoFile
   
	Buffer% = FREEFILE
	OPEN "MRDES.CFG" FOR INPUT AS #Buffer%

'   Clear the error flag

	ON ERROR GOTO 0

	FOR Counter% = 0 TO 6
		LINE INPUT #Buffer%, Work$
		Value%(Counter%) = VAL(LEFT$(Work$, 2))
	NEXT
	CLOSE
'============================================================================
'   Main (actually, only) loop.

Menu:
	DO

'   Clear screen, draw box.

		CLS
	   
		DrawMenuBox 1, 24, 5, 21, 78
	  
'   Print heading.  Center the strings.

		Cputs Heading$, 2
		Cputs Facility$, 3
		Cputs "MRDES II Configuration Utility version 1.30", 4
	   
		COLOR 7, 0
	   
'   Now options, and current values or examples of current values

		PutXY 6, 20, "1 - Normal Character Color"
		PutXY 7, 50, "("
		COLOR Value%(0), Value%(1)
		PRINT Value%(0);
	   
		COLOR 7, 0: PRINT ")"
	   
		PutXY 8, 20, "2 - Normal Background Color"
		PutXY 10, 20, "3 - Reverse Character Color"
		PutXY 11, 50, "("
		COLOR Value%(2), Value%(3)
		PRINT Value%(2);
		COLOR 7, 0: PRINT ")"
		PutXY 12, 20, "4 - Reverse Background Color"
		PutXY 14, 20, "5 - Printer   (Yes/No)"
		PutXY 14, 60, MID$("No Yes", Value%(4) * 3 + 1, 3)
	   
		PutXY 16, 20, "6 - Data Transfer"
		PutXY 16, 60, MID$("None    Rfl 5455 Knopp   BarcodeRfl 5800 ", Value%(5) * 8 + 1, 8)
		PutXY 17, 21, "(None/Rfl 5455/Knopp/Barcode/Rfl 5800)"

'   Tracking list cross check

		PutXY 19, 20, "7 - Tracking list report if check OK"
		IF Value%(6) = 0 THEN
			PutXY 19, 60, "Yes"
		ELSE
			PutXY 19, 60, "No"
		END IF


'   And the exits.  8 is meant as an "oops" escape.

		PutXY 22, 20, "8 - Exit without save"
		PutXY 23, 20, "9 - Save new values/Exit"

'   Menu is printed.  Get operator selection

		Select$ = Get1Key$("123456789", "Press number to select item")

		SELECT CASE Select$

			CASE "1"                ' Set normal character color
				ToggleValue Value%(0), 15

			CASE "2"                ' Set normal background color
				ToggleValue Value%(1), 7

			CASE "3"                ' Set reverse character color
				ToggleValue Value%(2), 15

			CASE "4"                ' Set reverse background color
				ToggleValue Value%(3), 7

			CASE "5"                ' Toggle Printer
				Value%(4) = ABS(Value%(4) = 0)

			CASE "6"                ' Data transfer
				ToggleValue Value%(5), 4
		   
			CASE "7"
				Value%(6) = ABS(Value%(6) = 0)

			CASE "9"
				CLS

'   At this point, we either write the new file, or just exit.
   
				ON ERROR GOTO Split
				Buffer% = FREEFILE
				OPEN "MRDES.CFG" FOR OUTPUT AS #Buffer%
				PRINT #Buffer%, MID$(STR$(Value%(0)), 2); " - Normal Character color"
				PRINT #Buffer%, MID$(STR$(Value%(1)), 2); " - Normal Background color"
				PRINT #Buffer%, MID$(STR$(Value%(2)), 2); " - Reverse Character color"
				PRINT #Buffer%, MID$(STR$(Value%(3)), 2); " - Reverse Background color"
				PRINT #Buffer%, MID$(STR$(Value%(4)), 2); " - Printer Selection"
				PRINT #Buffer%, MID$(STR$(Value%(5)), 2); " - Data Transfer"
				PRINT #Buffer%, MID$(STR$(Value%(6)), 2); "Cross check report if check ok"
			CLOSE
		END SELECT
  
	LOOP UNTIL INSTR("89", Select$) > 0
	RUN "MRDES"

	END

'   Error trap, in this case, just sets everything to defaults

NoFile:
	Value%(0) = 7   ' White
	Value%(1) = 0   ' Black
	Value%(2) = 0   ' Black
	Value%(3) = 7   ' White
	Value%(4) = 0   ' No printer
	Value%(5) = 0   ' No data transfer
	Value%(6) = 0   ' Report cross check
	RESUME Menu

Split:
	END
'****************************************************************************
'   End of MRSCFG.BAS main module
'****************************************************************************

'****************************************************************************
'   Increments value passed in Value%.  Sets to zero if greater than value
'   passed in Max%.
'============================================================================
SUB ToggleValue (Value%, Max%)
'----------------------------------------------------------------------------
	Value% = Value% + 1
	IF Value% > Max% THEN
		Value% = 0
	END IF
'****************************************************************************
END SUB
