'============================================================================
'   Define record type(s)
'----------------------------------------------------------------------------
'   Tracking data record type
'   This looks at the tracking list record as it is received from Enable
	TYPE RepairData
		Serial AS STRING * 9        ' Device serial number
		Code AS STRING * 5          ' Device code
		Company AS STRING * 1       ' Company code (use second char)
		Mfr AS STRING * 2           ' AEP Mfr code
		Ac AS STRING * 2
	END TYPE
'----------------------------------------------------------------------------
'   File access
	TYPE InfoRecord
		NumberOfRecords AS INTEGER  ' Number of records in the file
		Signature AS STRING * 20    ' TRDES machine signature
	END TYPE
	TYPE DataRecord
		Buffer AS STRING * 1000     ' 1000 characters in TRDES record
	END TYPE
'============================================================================
'   Declare procedures
'----------------------------------------------------------------------------
DECLARE FUNCTION AllCheck% (x$, Check$)
DECLARE FUNCTION CheckDate% (Value$)
DECLARE FUNCTION CheckDivision% (V$)
DECLARE FUNCTION CheckField% (Value$, Accept$)
DECLARE FUNCTION CheckRegion% (V$)
DECLARE SUB CheckRepairList (Value$(), Mfg%, Mbg%, Dfg%, Dbg%)
DECLARE SUB Cputs (Message$, Row%)
DECLARE SUB ErrMsg (Msg$, ClrFlag%)
DECLARE FUNCTION Get1Key$ (Accept$, Prompt$)
DECLARE SUB MfrLookup (Value$, Vlen%, Eflag%, Estring$)
DECLARE FUNCTION OneDec% (Look$)
DECLARE SUB PrtScrn (Prow%(), Pcol%(), Prompt$(), Arow%(), Acol%(), Value$(), Count%, Company$, Facility$, Mfg%, Mbg%, Dfg%, Dbg%, Number%)
DECLARE SUB PutXY (Row%, Column%, Message$)
DECLARE SUB RetireLookup (Value$, Vlen%, Eflag%, Estring$)
'============================================================================
'   $INCLUDE: 'RTDIMCOM.INC'
	DIM SHARED DataRec AS DataRecord, Info AS InfoRecord
'============================================================================
	Programmer$ = "K.W.Kroninger"
'   Version check
	IF COMMAND$ <> "" AND LEFT$(UCASE$(LTRIM$(COMMAND$)), 1) = "V" THEN
		PRINT "Reactiformer Data Check Utility - Version 9.00"
		END
	END IF
'============================================================================
'   Enable error checking
'----------------------------------------------------------------------------
	ON ERROR GOTO NetProblem
'============================================================================
'   Set a couple of flags

	ClrFlag% = 0    ' Screen has not been cleared
	ErrFlag% = 0    ' No errors found yet

'   Get the screen color right

	COLOR Mfg%, Mbg%
'============================================================================
'   Set up check strings
	All$ = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789- "
	Num$ = "0123456789"
	NumSp$ = "0123456789 "
	Alpha$ = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
	NumPer$ = "0123456789 ."
'============================================================================

'   Check code.  Check for all character positions filled with acceptable
'   characters and the first character is required to be a "P".
   
	IF AllCheck%(Value$(1), Alpha$ + Num$) = 1 OR LEFT$(Value$(1), 1) <> "P" THEN
		ErrMsg "Error - Code", ClrFlag%
	END IF

'   Check activity code.  Format, then check for acceptable values

	Value$(2) = RIGHT$(STRING$(Alen%(2), "0") + LTRIM$(RTRIM$(Value$(2))), Alen%(2))
	IF Value$(2) <> "01" AND Value$(2) <> "04" THEN
		ErrMsg "Error - Acvivity Code", ClrFlag%
	END IF

'   Check Serial Number.  Remove leading zeros, make sure there's at least
'   one character.

	Value$(3) = LTRIM$(RTRIM$(Value$(3)))
   
	WHILE LEN(Value$(3)) > 1 AND LEFT$(Value$(3), 1) = "0"
		Value$(3) = MID$(Value$(3), 2)
	WEND
   
	Value$(3) = LEFT$(LTRIM$(RTRIM$(Value$(3))) + SPACE$(Alen%(3)), Alen%(3))
   
	IF AllCheck%(Value$(3), All$) = 1 OR Value$(3) = SPACE$(Alen%(3)) THEN
		ErrMsg "Error - Serial Number", ClrFlag%
	END IF

'   Check Mfr with list of acceptable codes.
   
	MfrLookup Value$(4), Alen%(4), Flag%, " "

	IF AllCheck%(Value$(4), Alpha$ + Num$) OR Flag% THEN
		ErrMsg "Error - Manufacturer", ClrFlag%
	END IF

'   Check Dielectric Check.  Actually, just jam in a Y.
   
	Value$(9) = "Y"

'   Check Region code against lookup table for compliance with MDS.

	Value$(10) = LEFT$(UCASE$(Value$(10)) + SPACE$(Alen%(10)), Alen%(10))
 
	IF CheckRegion%(Value$(10)) THEN
		ErrMsg "Error - Region", ClrFlag%
	END IF

'   Check Tester.  Check for letters and minimum length of 3
   
	IF AllCheck%(Value$(11), Alpha$) = 1 OR LEN(LTRIM$(RTRIM$(Value$(11)))) < 3 THEN
		ErrMsg "Error - Tester ID", ClrFlag%
	END IF

'   Check Retirement Reason Code with table
   
	Flag% = 0
   
	RetireLookup Value$(12), Alen%(12), Flag%, " "
   
	IF Flag% THEN
		ErrMsg "Error - Retirement Reason Code", ClrFlag%
	END IF

'   Acceptable field
   
	IF AllCheck%(Value$(14), "YN") THEN
		ErrMsg "Error - Acceptable Flag", ClrFlag%
	END IF

'   Check Test Date
   
	IF CheckDate%(Value$(17)) = 1 THEN
		ErrMsg "Error - Date Received", ClrFlag%
	END IF

'   Price
   
	IF CheckField%(Value$(18), NumPer$ + "$") THEN
		ErrMsg "Error - Price", ClrFlag%
	END IF

'   Check damage and defect codes

	FOR x% = 19 TO 24
		IF Value$(x%) <> SPACE$(Alen%(x%)) THEN
			Value$(x%) = RIGHT$("00" + LTRIM$(RTRIM$(Value$(x%))), Alen%(x%))
			IF AllCheck%(Value$(x%), Num$) = 1 THEN
				IF x% < 22 THEN
					ErrMsg "Error - Damage code", ClrFlag%
				ELSE
					ErrMsg "Error - Defect code", ClrFlag%
				END IF
			END IF
		END IF
	NEXT

'   Check for same damage/defect code entered more then once

	IF Value$(19) = "  " AND Value$(20) <> "  " THEN
		SWAP Value$(19), Value$(20)
	END IF
   
	IF Value$(20) = "  " AND Value$(21) <> "  " THEN
		SWAP Value$(20), Value$(21)
		IF Value$(19) = "  " AND Value$(20) <> "  " THEN
			SWAP Value$(19), Value$(20)
		END IF
	END IF
   
	IF Value$(20) <> "  " AND Value$(19) = Value$(20) THEN
		ErrMsg "Duplicate damage codes", ClrFlag%
	END IF
   
	IF Value$(21) <> "  " AND (Value$(19) = Value$(21) OR Value$(20) = Value$(21)) THEN
		ErrMsg "Duplicate damage codes", ClrFlag%
	END IF
	   
	IF Value$(22) = "  " AND Value$(23) <> "  " THEN
		SWAP Value$(22), Value$(23)
	END IF
   
	IF Value$(23) = "  " AND Value$(24) <> "  " THEN
		SWAP Value$(23), Value$(24)
		IF Value$(22) = "  " AND Value$(23) <> "  " THEN
			SWAP Value$(22), Value$(23)
		END IF
	END IF
   
	IF Value$(23) <> "  " AND Value$(22) = Value$(23) THEN
		ErrMsg "Duplicate defect codes", ClrFlag%
	END IF
   
	IF Value$(24) <> "  " AND (Value$(22) = Value$(24) OR Value$(23) = Value$(24)) THEN
		ErrMsg "Duplicate defect codes", ClrFlag%
	END IF
	  
'============================================================================

'   Done with checks. Write it?
   
	IF ClrFlag% = 1 THEN
		GOTO WaitForUser
	END IF

'============================================================================
'   Now check repair list

	IF LTRIM$(RTRIM$(Value$(2))) = "" AND ClrFlag% = 0 THEN
		CheckRepairList Value$(), Mfg%, Mbg%, Dfg%, Dbg%
	END IF

'============================================================================

'   If it's check only, just return
   
	IF Flag% = 1 THEN
		CallPrg% = 2
		CHAIN "Rtdata"
	END IF
'============================================================================
'   All checks passed.  Write it?
   
	PrtScrn Prow%(), Pcol%(), Prompt$(), Arow%(), Acol%(), Value$(), 40, Company$, Facility$, Mfg%, Mbg%, Dfg%, Dbg%, 24
   
	COLOR Mfg%, Mbg%
	x$ = Get1Key$("YN", "Data checks OK.  Write it? (Y/N)")

	IF x$ = "N" THEN
		CallPrg% = 2
		CHAIN "Rtdata"
	END IF
'============================================================================
'   Put the data into the buffer, and write the record
'----------------------------------------------------------------------------
'   Establish string
	Buffer$ = SPACE$(1000)
'----------------------------------------------------------------------------
'   ID record type

'   Record types are as follows:
'       1=Current Transformer
'       3=Potential Transformer
'       6=Reactiformer
'       7=Metering Outfit
'----------------------------------------------------------------------------
	MID$(Buffer$, 1, 1) = "6"
	MID$(Buffer$, 2, Alen%(1)) = Value$(1)      '   Device Code
	MID$(Buffer$, 991, Alen%(2)) = Value$(2)    '   Activity Code
	MID$(Buffer$, 301, Alen%(3)) = Value$(3)    '   Serial Number
	MID$(Buffer$, 28, Alen%(4)) = Value$(4)     '   Mfg
	MID$(Buffer$, 33, Alen%(5)) = Value$(5)     '   Type
	MID$(Buffer$, 329, Alen%(6)) = Value$(6)    '   Volts
	MID$(Buffer$, 66, Alen%(9)) = Value$(9)     '   Dielectric Check
	MID$(Buffer$, 901, Alen%(10)) = Value$(10)  '   Division
	MID$(Buffer$, 71, Alen%(11)) = Value$(11)   '   Tester
	MID$(Buffer$, 327, Alen%(12)) = Value$(12)  '   Retirement Reason
	MID$(Buffer$, 75, Alen%(13)) = Value$(13)   '   Remarks
	MID$(Buffer$, 133, Alen%(14)) = Value$(14)  '   Acceptable unit
	MID$(Buffer$, 125, Alen%(17)) = Value$(17)  '   Test Date
	MID$(Buffer$, 277, Alen%(18)) = Value$(18)  '   Price
	MID$(Buffer$, 601, Alen%(19)) = Value$(19)  '   Damage code 1
	MID$(Buffer$, 603, Alen%(20)) = Value$(20)  '   Damage code 2
	MID$(Buffer$, 605, Alen%(21)) = Value$(21)  '   Damage code 3
	MID$(Buffer$, 607, Alen%(22)) = Value$(22)  '   Defect code 1
	MID$(Buffer$, 609, Alen%(23)) = Value$(23)  '   Defect code 2
	MID$(Buffer$, 611, Alen%(24)) = Value$(24)  '   Defect code 3
	MID$(Buffer$, 993, 8) = TIME$               '   Time Stamp
'============================================================================
'   Write the record to the disk
	Df% = FREEFILE
	OPEN FileName$ FOR RANDOM AS #Df% LEN = LEN(DataRec)
	GET #Df%, 1, Info
	RecNum% = Info.NumberOfRecords
	DataRec.Buffer = Buffer$

'   If we're editing a record, put it back in place.  If it's a new record,
'   append it to the file.

	IF EdRec% > 0 THEN
		RecNum% = EdRec%
	ELSE
		RecNum% = RecNum% + 1
	END IF
	PUT #Df%, RecNum% + 1, DataRec
	IF EdRec% = 0 THEN
		Info.NumberOfRecords = RecNum%
		Info.Signature = "TRDES"
		PUT #Df%, 1, Info
	END IF
	CLOSE #Df%
'----------------------------------------------------------------------------

'   And exit to the data input
   
	IF EdRec% > 0 THEN
		CallPrg% = 1
	ELSE
		CallPrg% = 4
	END IF
	CHAIN "Rtdata"
'----------------------------------------------------------------------------

'   It's bad.

WaitForUser:
	x$ = Get1Key$(CHR$(13), "Press ENTER to return to edit screen")
  
Split:
	CallPrg% = 2
	CHAIN "Rtdata"
'============================================================================
'   Network Problems

NetProblem:
	IF UCASE$(ENVIRON$("DEBUG")) = "YES" THEN
	   
		Junk$ = Get1Key$(CHR$(13) + CHR$(27), STR$(ERR))
   
	ELSE
	  
'   $INCLUDE: 'c:\mrswork\neterror.inc'
	   
	END IF
   
	IF Junk$ = CHR$(27) THEN
		RESUME Split
	ELSE
		RESUME
	END IF

'   End of error traps
'****************************************************************************
'   End of RTCHECK main module
'****************************************************************************
