'   July, 1996

'       Added code for all AEP Region Codes
'===========================================================================
'   This routine is used to input Reactiformer Test and record data.
'============================================================================
'   Declare procedures
'----------------------------------------------------------------------------
DECLARE SUB ACode ()
DECLARE SUB Cputs (Message$, Row%)
DECLARE SUB DamageCodes ()
DECLARE SUB DefectCodes ()
DECLARE SUB DisplayKeys (Xfer%, Mfg%, Mbg%, Dfg%, Dbg%)
DECLARE SUB FuncKeys ()
DECLARE FUNCTION Get1Key$ (Accept$, Prompt$)
DECLARE FUNCTION GetVSelection$ (Length%, FileName$, Select$())
DECLARE SUB MoveFieldPointer (Ptr%, Work%(), Vline%())
DECLARE SUB MfrCode ()
DECLARE SUB PrtScrn (Prow%(), Pcol%(), Prompt$(), Arow%(), Acol%(), Value$(), Count%, Company$, Facility$, Mfg%, Mbg%, Dfg%, Dbg%, Number%)
DECLARE SUB PutXY (Row%, Column%, Message$)
DECLARE SUB RetireCodes ()
'============================================================================
'   $INCLUDE: 'RTDIMCOM.INC'
'============================================================================
	Programmer$ = "K.W.Kroninger"
'   Version check
	IF COMMAND$ <> "" AND LEFT$(UCASE$(LTRIM$(COMMAND$)), 1) = "V" THEN
		PRINT "Reactiformer Data Entry/Edit Utility - Version 9.00"
		END
	END IF
'============================================================================
'   Turn on error trapping
'----------------------------------------------------------------------------
	ON ERROR GOTO NetProblem
'===========================================================================
'   If returning from RtCheck, skip initialization code
   
	IF CallPrg% = 2 THEN
		GOTO SkipInit
	END IF
   
	IF EdRec% > 0 AND CallPrg% = 1 THEN
		CHAIN "RTMENU"
	END IF
   
	IF EdRec% > 0 THEN
		GOTO SkipInit
	END IF
   
	IF CallPrg% = 4 THEN
		GOTO PartInit
	END IF

'   Initialize Value Strings

	FOR x% = 1 TO 24
		IF Arow%(x%) > 0 THEN
			Value$(x%) = SPACE$(Alen%(x%))
		END IF
	NEXT
   
	Value$(1) = LEFT$("P" + SPACE$(Alen%(1)), Alen%(1))
   
	Value$(17) = LEFT$(DATE$, 2) + MID$(DATE$, 4, 2) + RIGHT$(DATE$, 4)

PartInit:
   
	Value$(3) = SPACE$(Alen%(3))            ' Serial number
	Value$(9) = "Y"                         ' Dielectric check flag
	Value$(12) = SPACE$(Alen%(12))          ' Retirement reason code
	Value$(13) = SPACE$(Alen%(13))          ' Remarks
	Value$(14) = "Y"                        ' Acceptable unit
   
	FOR x% = 19 TO 24
		Value$(x%) = SPACE$(Alen%(x%))
	NEXT

SkipInit:

	PrtScrn Prow%(), Pcol%(), Prompt$(), Arow%(), Acol%(), Value$(), 24, Company$, Facility$, Mfg%, Mbg%, Dfg%, Dbg%, 24
   
	DisplayKeys 0, Mfg%, Mbg%, Dfg%, Dbg%

'   Input loop

	Ptr% = 1
	x$ = UCASE$(INKEY$)
	LOCATE Arow%(1), Acol%(1) + 1, 1, 0, 7

	Pointer% = 2

WaitForInput:

'   Get the input

	DO
		x$ = UCASE$(INKEY$)
	LOOP WHILE x$ = ""

	IF LEN(x$) > 1 THEN
		SELECT CASE ASC(RIGHT$(x$, 1))
			CASE 59             ' F1 - Write record to the disk
				Flag% = 0
				CallPrg% = 1
				CHAIN "RTCHECK"
			CASE 60             ' F2 - Return to menu
				GOSUB Escape
			CASE 63, 80         ' F5 - Next field
				GOSUB MoveToNext
			CASE 15, 64, 72     ' F6 - Prev field
				GOSUB MoveToPrev
			CASE 65, 77         ' F7 - Move cursor forward
				LOCATE , , 0
				Pointer% = Pointer% + 1
				IF Pointer% > Alen%(Ptr%) THEN
					MoveFieldPointer Ptr%, Fwd%(), Arow%()
					Pointer% = 1
				END IF
				LOCATE Arow%(Ptr%), Acol%(Ptr%) + Pointer% - 1, 1, 0, 7
				GOTO WaitForInput
			CASE 66, 75           ' F8 - Move cursor backward
				GOSUB MoveBwd
			CASE 67, 71           ' F9 - Cursor to first field
				GOSUB HomeC
			CASE 68               ' F10 - Help
				COLOR Mfg%, Mbg%
				CLS
				SELECT CASE Ptr%
					CASE 2
						Temp$ = GetVSelection$(Alen%(Ptr%), "ACTIVITY", Select$())
						IF LTRIM$(RTRIM$(Temp$)) <> "" THEN
							Value$(Ptr%) = LEFT$(LTRIM$(RTRIM$(Temp$)) + SPACE$(Alen%(Ptr%)), Alen%(Ptr%))
						END IF
					CASE 4
						Temp$ = GetVSelection$(Alen%(Ptr%), "MFR", Select$())
						IF LTRIM$(RTRIM$(Temp$)) <> "" THEN
							Value$(Ptr%) = LEFT$(LTRIM$(RTRIM$(Temp$)) + SPACE$(Alen%(Ptr%)), Alen%(Ptr%))
						END IF
					CASE 10
						Temp$ = GetVSelection$(Alen%(Ptr%), "REGION", Select$())
						IF LTRIM$(RTRIM$(Temp$)) <> "" THEN
							Value$(Ptr%) = LEFT$(LTRIM$(RTRIM$(Temp$)) + SPACE$(Alen%(Ptr%)), Alen%(Ptr%))
						END IF
					CASE 12
						RetireCodes
					CASE 19 TO 21
						DamageCodes
					CASE 22 TO 24
						DefectCodes
					CASE ELSE
						FuncKeys
				END SELECT

				PrtScrn Prow%(), Pcol%(), Prompt$(), Arow%(), Acol%(), Value$(), 24, Company$, Facility$, Mfg%, Mbg%, Dfg%, Dbg%, 24
  
				DisplayKeys 0, Mfg%, Mbg%, Dfg%, Dbg%
			   
				LOCATE Arow%(Ptr%), Acol%(Ptr%), 1, 0, 7
			   
				GOTO WaitForInput

		END SELECT
	END IF

'   Perform error check on Esc key
   
	IF x$ = CHR$(27) THEN
		Flag% = 1
		CallPrg% = 1
		CHAIN "RTCHECK"
	END IF

'   Enter = F5
   
	IF x$ = CHR$(9) OR x$ = CHR$(13) THEN
		GOSUB MoveToNext
	END IF

'   BackSpace = F8
   
	IF x$ = CHR$(8) THEN
		GOSUB MoveBwd
	END IF

'   Acceptable input?
   
	IF ASC(x$) < 32 OR ASC(x$) > 90 THEN
		GOTO WaitForInput
	END IF

'   Acceptable input.  Process it.
   
'   Store the input
   
	MID$(Value$(Ptr%), Pointer%, 1) = x$

'   Bump pointer
   
	Pointer% = Pointer% + 1

'   Put it on the screen
   
	LOCATE , , 0
	PRINT x$;

	IF Pointer% <= Alen%(Ptr%) THEN
		LOCATE , , 1, 0, 7
	ELSE
		MoveFieldPointer Ptr%, Fwd%(), Arow%()
		Pointer% = 1
		LOCATE Arow%(Ptr%), Acol%(Ptr%), 1, 0, 7
	END IF
	GOTO WaitForInput

'   End of input loop

'============================================================================
'   Subroutines begin here
'----------------------------------------------------------------------------
'   Function key 2 - Return to menu
Escape:
	COLOR Mfg%, Mbg%
	x$ = Get1Key$("YN", "Are you sure you want to return to the menu?(Y/N)")
	IF x$ = "Y" THEN
		CallPrg% = 1
		CHAIN "RTMENU"
	END IF
   
	PrtScrn Prow%(), Pcol%(), Prompt$(), Arow%(), Acol%(), Value$(), 24, Company$, Facility$, Mfg%, Mbg%, Dfg%, Dbg%, 24
   
	DisplayKeys 0, Mfg%, Mbg%, Dfg%, Dbg%
   
	GOTO HomeC
'----------------------------------------------------------------------------
'   Function key 5 - Move to next data entry field
MoveToNext:
	COLOR Dfg%, Dbg%
	PutXY Arow%(Ptr%), Acol%(Ptr%), Value$(Ptr%)
	MoveFieldPointer Ptr%, Fwd%(), Arow%()
	GOTO MovePointer
'----------------------------------------------------------------------------
'   Function key 6 - Move to previous data entry field
MoveToPrev:
	COLOR Dfg%, Dbg%
	PutXY Arow%(Ptr%), Acol%(Ptr%), Value$(Ptr%)
	MoveFieldPointer Ptr%, Bwd%(), Arow%()
'----------------------------------------------------------------------------
MovePointer:
	LOCATE , , 0
	SELECT CASE Ptr%
		CASE 2
			IF Value$(Ptr%) = SPACE$(Alen%(Ptr%)) THEN
				Temp$ = GetVSelection$(Alen%(Ptr%), "ACTIVITY", Select$())
				IF LTRIM$(RTRIM$(Temp$)) <> "" THEN
					Value$(Ptr%) = LEFT$(LTRIM$(RTRIM$(Temp$)) + SPACE$(Alen%(Ptr%)), Alen%(Ptr%))
				END IF
			END IF
		CASE 4
			IF Value$(Ptr%) = SPACE$(Alen%(Ptr%)) THEN
				Temp$ = GetVSelection$(Alen%(Ptr%), "MFR", Select$())
				IF LTRIM$(RTRIM$(Temp$)) <> "" THEN
					Value$(Ptr%) = LEFT$(LTRIM$(RTRIM$(Temp$)) + SPACE$(Alen%(Ptr%)), Alen%(Ptr%))
				END IF
			END IF
		CASE 10
			IF Value$(Ptr%) = SPACE$(Alen%(Ptr%)) THEN
				Temp$ = GetVSelection$(Alen%(Ptr%), "REGION", Select$())
				IF LTRIM$(RTRIM$(Temp$)) <> "" THEN
					Value$(Ptr%) = LEFT$(LTRIM$(RTRIM$(Temp$)) + SPACE$(Alen%(Ptr%)), Alen%(Ptr%))
				END IF
			END IF
	END SELECT
	COLOR Dfg%, Dbg%
	PutXY Arow%(Ptr%), Acol%(Ptr%), Value$(Ptr%)
	LOCATE Arow%(Ptr%), Acol%(Ptr%), 1, 0, 7
	Pointer% = 1
	RETURN WaitForInput
'----------------------------------------------------------------------------
'   Function key 8 - Move cursor backward
MoveBwd:
	LOCATE , , 0
	Pointer% = Pointer% - 1
	IF Pointer% > 0 THEN
		LOCATE Arow%(Ptr%), Acol%(Ptr%) + Pointer% - 1, 1, 0, 7
		RETURN WaitForInput
	END IF

'   Move to previous field.  Code is about the same as MOVETOPREV, except
'   that we wind up at the end of the field instead of the beginning.
   
	MoveFieldPointer Ptr%, Bwd%(), Arow%()
   
	LOCATE Arow%(Ptr%), Acol%(Ptr%) + Alen%(Ptr%) - 1, 1, 0, 7
	Pointer% = Alen%(Ptr%)
	RETURN WaitForInput
'----------------------------------------------------------------------------
'   F9 - Home cursor
HomeC:
	Ptr% = 1
	Pointer% = 1
	GOTO MovePointer
'============================================================================
'   Error traps
'----------------------------------------------------------------------------
'   Network Problems

NetProblem:
	IF UCASE$(ENVIRON$("DEBUG")) = "YES" THEN
	   
		Junk$ = Get1Key$(CHR$(13) + CHR$(27), STR$(ERR))
   
	ELSE
	   
'   $INCLUDE: 'c:\mrswork\neterror.inc'

	END IF
   
	IF Junk$ = CHR$(27) THEN
		RESUME WaitForInput
	ELSE
		RESUME
	END IF

'   End of error traps
'****************************************************************************
'   End of RTDATA main module
'****************************************************************************
