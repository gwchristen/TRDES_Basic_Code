'   Changed from lpt2 to lpt1

DIM SHARED D$, N1$, N2$, N3$, E$, Q$, W$, M1$, M2$, M3$
DECLARE SUB MESSAGE1 ()
DECLARE SUB MESSAGE2 ()
DECLARE SUB ENTRY1 ()
DECLARE SUB ENTRY2 ()
DECLARE SUB ENTRY3 ()
DECLARE SUB ENDOF ()
DECLARE SUB DOTHEJOB ()

Programmer$ = "R.W.Oehler"
E$ = CHR$(27)
OPEN "lpt1:" FOR OUTPUT AS #1

CLS
PRINT "HANDY LABEL PRINTER WRITTEN FOR SATO... DEFAULT IS lpt1 / V1.0"
PRINT

AGAIN:
    ENTRY1
    ENTRY2
    ENTRY3

HOLD1:
    MESSAGE1

HOLD:

D$ = INKEY$
IF UCASE$(D$) = "N" THEN
    CLS
    N1$ = ""
    N2$ = ""
    N3$ = ""
    GOTO AGAIN

ELSEIF UCASE$(D$) = "" THEN GOTO HOLD
END IF

REPEAT1:

CLS

    INPUT "HOW MANY LABELS DO YOU WANT TO PRINT"; G

W$ = LTRIM$(RTRIM$("L0202" + E$ + "M"))

FOR T = 1 TO G
   
    DOTHEJOB

NEXT T    'END OF ONLY DO LOOP IN THIS PROGRAM!

    ENDOF
    MESSAGE2

REHOLD:
D$ = INKEY$
IF UCASE$(D$) = "S" THEN
    CLS
    N1$ = ""
    N2$ = ""
    N3$ = ""
    GOTO AGAIN

ELSEIF UCASE$(D$) = "" THEN GOTO REHOLD
ELSEIF UCASE$(D$) = "Q" THEN END
ELSEIF UCASE$(D$) = "R" THEN GOTO HOLD1

END IF

CLS

END

SUB DOTHEJOB

PRINT #1, E$; "A";
PRINT #1, E$; "A3H0820V0001";    ' A3H SEEMS TO SET THE DEFAULT HORZ OFFSET
                              ' (820 LOOKS GOOD) "V000" SETS VERT. OFFSET
                              '  SET THE VERT AND HORZ OFFSET ON EACH LINE!
M1$ = LTRIM$(RTRIM$(N1$))
M2$ = LTRIM$(RTRIM$(N2$))
M3$ = LTRIM$(RTRIM$(N3$))

PRINT #1, E$; "H15"; E$; "V001"; E$; W$; M1$       ' PRINTS LINE 1 V IS VERT HEIGHT
PRINT #1, E$; "H15"; E$; "V040"; E$; W$; M2$       ' PRINTS LINE 2
PRINT #1, E$; "H15"; E$; "V082"; E$; W$; M3$       ' PRINTS LINE 3

PRINT #1, E$; "Q1";         ' Q SEEMS TO BE END OF MESSAGE
PRINT #1, E$; "Z";          ' Z SEEMS TO MAKE IT PRINT


END SUB

SUB ENDOF

'  THE FOLLOWING CODE SPITS OUT 2 LABELS WITH "END OF JOB" ON
'  NORMAL TERMINATION

Q$ = LTRIM$(RTRIM$("L0302" + E$ + "M*END OF JOB"))

PRINT #1, E$; "A";
PRINT #1, E$; "A3H0820V0001";    ' A3H SEEMS TO SET THE DEFAULT HORZ OFFSET
                                 ' (820 LOOKS GOOD) "V000" SETS VERT. OFFSET

PRINT #1, E$; "H15"; E$; "V030"; E$; Q$        ' PRINTS THE LINE OF TEXT
PRINT #1, E$; "Q1";         ' Q SEEMS TO BE END OF MESSAGE
PRINT #1, E$; "Z";          ' Z SEEMS TO MAKE IT PRINT
PRINT #1, E$; "A"; E$; "C"; E$; "Z"

CLS

END SUB

SUB ENTRY1

AGAIN1:
LOCATE 5, 16
PRINT "^^^^^^^^^^^^^^^^^^^^^"
LOCATE 3, 1
PRINT "     ENTER THE FIRST LINE OF TEXT "
INPUT "21 DIGITS MAX! ", N1$

IF LEN(N1$) > 21 THEN
    CLS
    PRINT
    PRINT
   
    PRINT "YOU HAVE ENTERED MORE THAN 21 DIGITS."
    PRINT "PLEASE RE-ENTER..."
    N1$ = ""
    GOTO AGAIN1

END IF
CLS

END SUB

SUB ENTRY2

AGAIN2:
LOCATE 5, 16
PRINT "^^^^^^^^^^^^^^^^^^^^^"
LOCATE 3, 1
PRINT "       ENTER THE SECOND LINE OF TEXT"
INPUT "21 DIGITS MAX! ", N2$

IF LEN(N2$) > 21 THEN
    CLS
    PRINT
    PRINT
  
    PRINT "YOU HAVE ENTERED MORE THAN 21 DIGITS."
    PRINT "PLEASE RE-ENTER LINE 2..."
    N2$ = ""
    GOTO AGAIN2
END IF
  
CLS

END SUB

SUB ENTRY3
'COLOR 7, 1

AGAIN3:
LOCATE 5, 16
PRINT "^^^^^^^^^^^^^^^^^^^^^"
LOCATE 3, 1
PRINT "     ENTER THE THIRD LINE OF TEXT    - I DON'T LIKE COMMAS!"
LOCATE 4, 1: INPUT "21 DIGITS MAX! ", N3$

IF LEN(N3$) > 21 THEN
    CLS
    PRINT "YOU HAVE ENTERED MORE THAN 21 DIGITS."
    PRINT "PLEASE RE-ENTER..."
    N3$ = ""
    GOTO AGAIN3

END IF

END SUB

SUB MESSAGE1

CLS
PRINT "YOUR MESSAGE WILL APPEAR AS:"
PRINT
PRINT
LOCATE 5, 10: PRINT N1$
LOCATE 6, 10: PRINT N2$
LOCATE 7, 10: PRINT N3$
PRINT
PRINT
PRINT "PRESS N TO REDO FROM START OR ANY OTHER KEY TO CONTINUE"

END SUB

SUB MESSAGE2

PRINT
PRINT "YOUR JOB IS COMPLETE  -  WOULD YOU LIKE TO:"
PRINT "                         Q- QUIT"
PRINT "                         R- REPEAT THE JOB WITH THE SAME LABEL"
PRINT "                         S- START THIS PROGRAM AGAIN FROM THE TOP"
PRINT
PRINT "PLEASE PICK ONE  (Q-R-S)"

END SUB
