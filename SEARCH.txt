'============================================================================
'   Dimension array
'----------------------------------------------------------------------------
    DIM SHARED InFile$, Value$(100)
'============================================================================
    Programmer$ = "K.W.Kroninger"
'   Network error trap
    ON ERROR GOTO NetProblem
'----------------------------------------------------------------------------
'   If input filespec is on command line, use it, otherwise prompt for it.
'----------------------------------------------------------------------------
    CLS
    InFile$ = LTRIM$(RTRIM$(COMMAND$))
    IF InFile$ = "" THEN
        LINE INPUT "Input Filespec:"; InFile$
        InFile$ = LTRIM$(RTRIM$(InFile$))
        IF InFile$ = "" THEN
            END
        END IF
    END IF
 
    DO
        x$ = Menu$
        SELECT CASE x$
            CASE "EXIT", "QUIT"
                EXIT DO
            CASE "HELP"
                HelpScreen
            CASE "ALL"
                InFile% = FREEFILE
                CLS
                OPEN InFile$ FOR INPUT AS #InFile%
                Count% = 0
                DO
                    LINE INPUT #InFile%, Work$
                    IF LEN(RTRIM$(LTRIM$(Work$))) > 50 THEN
                        PRINT RTRIM$(MID$(Work$, 39, 26))
                        Count% = Count% + 1
                        IF Count% = 22 THEN
                            y$ = Get1Key$(CHR$(13) + CHR$(27), "Press Enter to continue, Esc to Quit")
                            Count% = 0
                            IF y$ = CHR$(27) THEN
                                EXIT DO
                            END IF
                            CLS
                        END IF
                    END IF
                LOOP UNTIL EOF(InFile%)
                CLOSE
                IF Count% > 0 THEN
                    y$ = Get1Key$(CHR$(13) + CHR$(27) + CHR$(32), "Press Enter/Esc/Space to return to menu")
                END IF
            CASE "CHANGE"
                Change
            CASE ELSE
                y$ = ""
                Found% = 0
                InFile% = FREEFILE
                OPEN InFile$ FOR INPUT AS #InFile%
                DO
                    y$ = INKEY$
                    LINE INPUT #InFile%, Work$              ' Get a record
                    IF LEN(LTRIM$(RTRIM$(Work$))) > 50 THEN
                        Serial$ = RTRIM$(MID$(Work$, 39, 26))
                        IF LEFT$(Serial$, LEN(x$)) = x$ THEN
                            Found% = 1
                            ParseDataRecord (Work$)
                            SCREEN , , 1, 1
                            PrintDataRecord
                            y$ = Get1Key$(CHR$(13) + CHR$(27), "Enter to search for more records, Esc to quit")
                            SCREEN , , 0, 0
                        END IF
                    END IF
                LOOP UNTIL EOF(InFile%) OR y$ = CHR$(27)
                CLOSE
                IF y$ <> CHR$(27) AND Found% = 0 THEN
                    Cputs SPACE$(30), 19
                    Cputs "No Record Found", 19
                    y$ = Get1Key$(CHR$(13), "Press Enter to continue")
                END IF
        END SELECT
    LOOP

Split:
    END

'============================================================================
'   Error traps
'----------------------------------------------------------------------------
'   Network Problems

NetProblem:
    IF UCASE$(ENVIRON$("DEBUG")) = "YES" THEN
        Junk$ = Get1Key$(CHR$(13) + CHR$(27), STR$(ERR))
    ELSE
        x& = TIMER + 10
        DO
        Junk$ = INKEY$
        LOOP UNTIL Junk$ = CHR$(27) OR x& < TIMER
    END IF
    IF Junk$ = CHR$(27) THEN
        RESUME Split
    ELSE
        RESUME
    END IF

'****************************************************************************
'   Change information in record.
'============================================================================
SUB Change
'----------------------------------------------------------------------------
    x% = 4
    DIM Column%(x%), Row%(x%)
    CLS
    DrawMenuBox 1, 24, 3, 0, 78

    Cputs "Search Utility Change Procedure", 2
    PutXY 7, 20, "Serial number (required):"
    PutXY 9, 16, "New serial number (optional):"
    PutXY 11, 33, "Device Code:"
    PutXY 13, 27, "Manufacturer Code:"
    PutXY 15, 10, "You must enter a serial number. All records in the file"
    PutXY 16, 10, "with that serial number will be changed."
    PutXY 18, 10, "If you wish to change the serial number, device code, or"
    PutXY 19, 10, "manufacturer code, enter the new data in the fields.  If"
    PutXY 20, 10, "you leave a field blank, that data will not be changed."
    PutXY 21, 10, "When the data is entered, press F10 to do the change,"
    PutXY 22, 10, "or Esc to exit without changing data"
    PutXY 4, 5, "F10 = Input complete, do change."
    PutXY 4, 50, "Esc = Exit without change"
    Value$(1) = SPACE$(9)
    Value$(2) = SPACE$(9)
    Value$(3) = SPACE$(5)
    Value$(4) = SPACE$(2)
    COLOR 15, 4
    Row%(1) = 7
    Row%(2) = 9
    Row%(3) = 11
    Row%(4) = 13
    FOR x% = 1 TO 4
        Column%(x%) = 44
        PutXY Row%(x%), Column%(x%) + 1, Value$(x%)
    NEXT
    DataField% = 1
    Pointer% = 1
    DO
        LOCATE Row%(DataField%), Column%(DataField%) + Pointer%, 1
        DO
            x$ = INKEY$
        LOOP WHILE x$ = ""
        LOCATE , , 0
        IF LEN(x$) = 2 THEN
            SELECT CASE ASC(RIGHT$(x$, 1))
                CASE 68             ' F10 key pressed
                    x$ = CHR$(192)
                CASE 15             ' Shift/Tab
                    IF DataField% = 1 THEN
                        DataField% = 4
                    ELSE
                        DataField% = DataField% - 1
                    END IF
                    Pointer% = 1
                CASE 71             ' Home key
                    DataField% = 1
                    Pointer% = 1
                CASE 79             ' End key
                    DataField% = 4
                    Pointer% = 1
                CASE 72             ' Up arrow
                    IF DataField% = 1 THEN
                        DataField% = 4
                    ELSE
                        DataField% = DataField% - 1
                    END IF
                    Pointer% = 1
                CASE 80             ' Down arrow
                    IF DataField% = 4 THEN
                        DataField% = 1
                    ELSE
                        DataField% = DataField% + 1
                    END IF
                    Pointer% = 1
                CASE 75             ' Left arrow
                    IF Pointer% = 1 THEN
                        IF DataField% = 1 THEN
                            DataField% = 4
                        ELSE
                            DataField% = DataField% - 1
                        END IF
                        Pointer% = LEN(Value$(DataField%))
                    ELSE
                        Pointer% = Pointer% - 1
                    END IF
                CASE 77             ' Right arrow
                    IF Pointer% = LEN(Value$(DataField%)) THEN
                        IF DataField% = 4 THEN
                            DataField% = 1
                        ELSE
                            DataField% = DataField% + 1
                        END IF
                        Pointer% = 1
                    ELSE
                        Pointer% = Pointer% + 1
                    END IF
                CASE ELSE
            END SELECT
        ELSE
            IF x$ <> CHR$(27) AND x$ > CHR$(31) AND x$ < CHR$(127) THEN
                MID$(Value$(DataField%), Pointer%, 1) = UCASE$(x$)
                PutXY Row%(DataField%), Column%(DataField%) + Pointer%, x$
                Pointer% = Pointer% + 1
                IF Pointer% > LEN(Value$(DataField%)) THEN
                    Pointer% = 1
                    IF DataField% = 4 THEN
                        DataField% = 1
                    ELSE
                        DataField% = DataField% + 1
                    END IF
                END IF
            ELSE
                IF x$ = CHR$(9) OR x$ = CHR$(13) THEN
                    IF DataField% = 4 THEN
                        DataField% = 1
                    ELSE
                        DataField% = DataField% + 1
                    END IF
                    Pointer% = 1
                END IF
            END IF                  ' x$ it to be put in char position
        END IF                      ' len x$=1
        LOCATE Row%(DataField%), Column%(DataField%) + Pointer%, 1
    LOOP WHILE x$ <> CHR$(27) AND x$ <> CHR$(192)
    FOR x% = 1 TO 4
        Value$(x%) = LTRIM$(RTRIM$(Value$(x%)))
        IF Value$(x%) > "" THEN
            IF INSTR(Value$(x%), CHR$(32)) > 0 THEN
                Value$(x%) = ""
            END IF
        END IF
    NEXT
    
    IF Value$(1) = "" THEN
        EXIT SUB
    END IF
    IF Value$(2) = "" AND Value$(3) = "" AND Value$(4) = "" THEN
        EXIT SUB
    END IF

'   If we get here, there's some data to process


    y$ = ""
    x$ = RTRIM$(LTRIM$(x$))
    Found% = 0
    InFile% = FREEFILE
    OPEN InFile$ FOR INPUT AS #InFile%
    DO
        y$ = INKEY$
        LINE INPUT #InFile%, Work$              ' Get a record
        IF LEN(LTRIM$(RTRIM$(Work$))) > 50 THEN
            Serial$ = RTRIM$(MID$(Work$, 39, 26))
            IF LEFT$(Serial$, LEN(Value$(1))) = Value$(1) THEN
                Found% = 1
                ParseDataRecord (Work$)
                SCREEN , , 1, 1
                PrintDataRecord
                y$ = Get1Key$(CHR$(13) + CHR$(27), "Enter to search for more records, Esc to quit")
                SCREEN , , 0, 0
            END IF
        END IF
    LOOP UNTIL EOF(InFile%) OR y$ = CHR$(27)
    CLOSE
    IF y$ <> CHR$(27) AND Found% = 0 THEN
        Cputs SPACE$(30), 19
        Cputs "No Record Found", 19
        y$ = Get1Key$(CHR$(13), "Press Enter to continue")
    END IF






'****************************************************************************
END SUB

'****************************************************************************
'   Checks Test$ to see that is in Accept$.  Prints string.  Adds error
'   message if all is not well.
'============================================================================

'----------------------------------------------------------------------------
    IF INSTR(Accept$, Test$) > 0 THEN
        PRINT Test$
    ELSE
        PRINT Test$ + " - Invalid data in field"
    END IF
'****************************************************************************
END SUB

'----------------------------------------------------------------------------

END SUB

'****************************************************************************
'   Parse Elements from Interface File Data Record
'============================================================================
SUB ParseDataRecord (Work$)
'----------------------------------------------------------------------------
    Work$ = Work$ + SPACE$(500)         ' Make string long enough to eliminate problems
    Value$(1) = MID$(Work$, 1, 1)       ' Transaction Code
    Value$(2) = MID$(Work$, 2, 2)       ' Company Code
    Value$(3) = MID$(Work$, 4, 2)       ' State Code
    Value$(4) = MID$(Work$, 6, 2)       ' Division Code
    Value$(5) = MID$(Work$, 8, 3)       ' Area Code
    Value$(6) = MID$(Work$, 11, 3)      ' Storeroom Number
    Value$(7) = MID$(Work$, 14, 2)      ' Receiving State Code
    Value$(8) = MID$(Work$, 16, 2)      ' Receiving Division Code
    Value$(9) = MID$(Work$, 18, 3)      ' Receiving Area Code
    Value$(10) = MID$(Work$, 21, 3)     ' Receiving Storeroom Code
    Value$(11) = MID$(Work$, 24, 2)     ' Status Code
    Value$(12) = MID$(Work$, 26, 5)     ' Purchase Code
    Value$(13) = MID$(Work$, 31, 1)     ' Previous Installed Indicator (Y/N)
    Value$(14) = MID$(Work$, 32, 2)     ' Device Manufacturer Code
    Value$(15) = MID$(Work$, 34, 5)     ' Device Code
    Value$(16) = MID$(Work$, 39, 26)    ' Device Serial Number
    Value$(17) = MID$(Work$, 65, 5)     ' Device Code (Accessory Device Only)
    Value$(18) = MID$(Work$, 70, 2)     ' Accessory Type Code (Accessory Device Only)
    Value$(19) = MID$(Work$, 72, 2)     ' Manufacturer Code (Accessory Device Only)
    Value$(20) = MID$(Work$, 74, 4)     ' Battery Check Year (Associated Device Only)
    Value$(21) = MID$(Work$, 78, 2)     ' Associated Manufacturer Code (Register Only)
    Value$(22) = MID$(Work$, 80, 5)     ' Associated Device Code (Register Only)
    Value$(23) = MID$(Work$, 85, 26)    ' Associated Device Serial Number (Register Only)
    Value$(24) = MID$(Work$, 111, 8)    ' Tester Identification
    Value$(25) = MID$(Work$, 119, 2)    ' Activity Type Code
    Value$(26) = MID$(Work$, 121, 2)    ' Test Description Code (FD=Field, SH=Shop)

'****************************************************************************
'   Decode and print data record
'============================================================================
SUB PrintDataRecord
'----------------------------------------------------------------------------
    CLS
   
    PrintaLine
  
    PRINT "MDS Transaction Code:" + TransactionCode$(Value$(1))
  
   
    RequiredData "Company Code:", Value$(2)
   
    LOCATE CSRLIN - 1, 25

    PRINT "Division Code:";
    IF LTRIM$(RTRIM$(Value$(4))) <> "" THEN
        PRINT DecodeDivision$(Value$(4))
    ELSE
        DataMissing
    END IF
  
    RequiredData "Area Code:", Value$(5)
  
    IF Value$(1) = "5" THEN
        RequiredData "Receiving State Code:", Value$(7)
        PRINT "Receiving Division Code:";
        IF LTRIM$(RTRIM$(Value$(8))) <> "" THEN
            PRINT DecodeDivision$(Value$(8))
        ELSE
            DataMissing
        END IF
 
        RequiredData "Receiving Area Code:", Value$(9)
    END IF
  
    IF Value$(1) = "6" OR LTRIM$(RTRIM$(Value$(11))) <> "" THEN
        PRINT "Status Code:";
            IF Value$(1) <> "6" OR LTRIM$(RTRIM$(Value$(11))) <> "" THEN
                PRINT Value$(11)
            ELSE
                DataMissing
            END IF
    END IF
  
    IF Value$(1) = "2" OR Value$(1) = "4" THEN
        RequiredData "Purchase Code:", Value$(12)
        PRINT "Previous Installed Indicator (Y/N):";
        CheckString "YN", Value$(13)
    END IF
  
    RequiredData "Device Manufacturer Code:", Value$(14)
  
    RequiredData "Device Code:", Value$(15)
  
    RequiredData "Device Serial Number:", Value$(16)
  
'   If there's no test transaction, this record is done.

    IF Value$(1) > "2" THEN
        EXIT SUB
    END IF
  
    IF LTRIM$(RTRIM$(Value$(24))) <> "" THEN
        PRINT "Tester Identification:" + Value$(24)
    END IF

    IF CSRLIN > 23 THEN
        Continue
    END IF

    PRINT "Activity Type Code:";
    IF LTRIM$(RTRIM$(Value$(25))) <> "" THEN
        PRINT ActivityCode$(Value$(25))
    ELSE
        DataMissing
    END IF
  
    IF CSRLIN > 23 THEN
        Continue
    END IF

    PRINT "Test Date (MmDdCcYy):";
    IF LTRIM$(RTRIM$(Value$(27))) <> "" THEN
        Flag% = 0
        FOR c% = 1 TO LEN(Value$(27))
            IF INSTR("0123456789", MID$(Value$(27), c%, 1)) = 0 THEN
                Flag% = 1
            END IF
        NEXT
        IF Flag% = 0 THEN
            PRINT Value$(27)
        ELSE
            PRINT Value$(27) + " - Invalid data in field"
        END IF
    ELSE
        DataMissing
    END IF
   
    IF CSRLIN > 23 THEN
        Continue
    END IF

    OptionalData "Remarks Regarding Test:", LTRIM$(RTRIM$(Value$(29)))
   
    IF CSRLIN > 23 THEN
        Continue
    END IF

    IF LTRIM$(RTRIM$(Value$(30))) <> "" THEN
        PRINT "Retirement Reason:" + RetireCode$(Value$(30))
    END IF
  
    IF CSRLIN > 23 THEN
        Continue
    END IF

    PRINT "Repair Indicator (Y/N):";
    IF INSTR("YN", Value$(31)) > 0 THEN
        PRINT Value$(31)
    ELSE
        PRINT Value$(31) + " Invalid data in field"
    END IF
   
    LOCATE CSRLIN - 1, 30
  
    PRINT "Meter Stop Indicator (Y/N):";
    CheckString "YN", Value$(38)
   
    IF CSRLIN > 23 THEN
        Continue
    END IF

    IF LTRIM$(RTRIM$(Value$(32))) <> "" THEN
        PRINT "Damage Code One:" + DamageCode$(Value$(32))
    END IF
   
    IF CSRLIN > 23 THEN
        Continue
    END IF

    IF LTRIM$(RTRIM$(Value$(33))) <> "" THEN
        PRINT "Damage Code Two:" + DamageCode$(Value$(33))
    END IF
   
    IF CSRLIN > 23 THEN
        Continue
    END IF

    IF LTRIM$(RTRIM$(Value$(34))) <> "" THEN
        PRINT "Damage Code Three:" + DamageCode$(Value$(34))

    IF CSRLIN > 23 THEN

    END IF

    END IF
   
    IF CSRLIN > 23 THEN

    END IF

    IF LTRIM$(RTRIM$(Value$(36))) <> "" THEN

    END IF

'****************************************************************************
'   Prints identification, then either data or missing message
'============================================================================
SUB RequiredData (Id$, Value$)
'----------------------------------------------------------------------------
    PRINT Id$;
    IF LTRIM$(RTRIM$(Value$)) <> "" THEN
        PRINT Value$
    ELSE
        DataMissing
    END IF
'****************************************************************************
END SUB

'****************************************************************************
'   Decode retirement code
'============================================================================
FUNCTION RetireCode$ (Pointer$)
'----------------------------------------------------------------------------
    SELECT CASE Pointer$
        CASE "01"
            RetireCode$ = Pointer$ + " - Lost/Stolen"
        CASE "02"
            RetireCode$ = Pointer$ + " - Damaged"
        CASE "03"
            RetireCode$ = Pointer$ + " - Obsolete"
        CASE "04"
            RetireCode$ = Pointer$ + " - Defective"
        CASE "05"
            RetireCode$ = Pointer$ + " - Utility Plant Sold"
        CASE ELSE
            RetireCode$ = Pointer$ + " - Unknown retirement code"
    END SELECT
'****************************************************************************
END FUNCTION

'****************************************************************************
'   Decode MDS transaction code
'============================================================================
FUNCTION TransactionCode$ (Pointer$)
'----------------------------------------------------------------------------
    SELECT CASE Pointer$
        CASE "1"
            TransactionCode$ = Pointer$ + " - Test Record"
        CASE "2"
            TransactionCode$ = Pointer$ + " - Test/History Record"
        CASE "3"
            TransactionCode$ = Pointer$ + " - Physical Inventory Record"
        CASE "4"
            TransactionCode$ = Pointer$ + " - History Record"
        CASE "5"
            TransactionCode$ = Pointer$ + " - Transfer Record"
        CASE "6"
            TransactionCode$ = Pointer$ + " - Status Change Record"
        CASE "7"
            TransactionCode$ = Pointer$ + " - Battery Check Year Record"
        CASE ELSE
            TransactionCode$ = Pointer$ + " - Unknown transaction code"
    END SELECT
'****************************************************************************
END FUNCTION
