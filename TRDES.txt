'****************************************************************************
'   Calculate accuracy class from Ratio and Phase error
'   Replacement formula per Jack Carr - January, 1990
'   Minor mod in May, 1994, to make the same routine work for both current
'   and voltage transformers.  The only difference seems to be the
'   calculation of the correction factor using the ratio and phase error.
'============================================================================
FUNCTION AcCalc$ (Ratio$, Phase$, Alen%, Sign%)
'----------------------------------------------------------------------------
'   Beta is in minutes

    Beta = VAL(Phase$)

'   Calculate ratio correction factor from formula

    Ratio = VAL(Ratio$) / 100
   
'   Sign% must be either -1 or +1

    Tcf = Ratio + (Beta / 2600) * Sign%

'   Use the greater of what we calculated or the transformer ratio error

    IF ABS(1 - Tcf) > ABS(1 - Ratio) THEN
        AcClass = ABS(1 - Tcf) * 100
    ELSE
        AcClass = ABS(1 - Ratio) * 100
    END IF

'   Round to two decimal places

    IF AcClass = 0 THEN
        AcCalc$ = LEFT$(".00" + SPACE$(Alen%), Alen%)
        EXIT FUNCTION
    END IF
  
    AcClass = (INT(AcClass * 100 + .5)) / 100
    x$ = STR$(AcClass)

'   If there's no period, somthing's wrong.
'   Set return string to spaces and exit.

    IF INSTR(x$, ".") = 0 THEN
        AcCalc$ = SPACE$(Alen%)
        EXIT FUNCTION
    END IF

'   Remove all characters ahead of period

    x$ = LTRIM$(RTRIM$(x$))

'   Make this string 4 characters and put it in the proper place.

    AcCalc$ = LEFT$(x$ + SPACE$(Alen%), Alen%)
'****************************************************************************
END FUNCTION

'****************************************************************************
'   Prefixes sign character to string, if necessary
'============================================================================
FUNCTION AddSign$ (Work$, Alen%)
'----------------------------------------------------------------------------
    Work$ = LTRIM$(RTRIM$(Work$))
    IF INSTR("+-", LEFT$(Work$, 1)) = 0 THEN
        Work$ = "+" + Work$
    END IF

'   Make it the correct length

    AddSign$ = LEFT$(Work$ + SPACE$(Alen%), Alen%)
'****************************************************************************
END FUNCTION

'****************************************************************************
'   Checks ratio for value within span specified in Environment variable
'   RATIO.  Defaults to 2.00 if env variable is not set.  Check is disabled
'   if variable is set to 0.00 value.
'============================================================================
SUB CheckRatio (Value$, ClrFlag%, LineNumber$)
'----------------------------------------------------------------------------
 
'   If the field is blank, check ends.

    IF LTRIM$(RTRIM$(Value$)) = "" THEN
        EXIT SUB
    END IF
 
'   Check for override value in environment variable

    Allowed$ = LTRIM$(RTRIM$(ENVIRON$("RATIO")))
 
'   If override is there, use that information, otherwise use default.

    IF Allowed$ = "" THEN
        Allowed! = 2!
    ELSE
        Allowed! = VAL(Allowed$)
    END IF
 
'   Setting override to zero disables check.

    IF Allowed$ = "0.00" THEN
        EXIT SUB
    END IF
  
    IF ABS(VAL(Value$) - 100!) > Allowed! THEN
        ErrMsg LineNumber$ + " Ratio out of limits.", ClrFlag%
    END IF
'****************************************************************************
END SUB

'****************************************************************************
'   Open the Repair List file and find out how many records are in it.  The
'   odd length allows direct export from Enable database to a text file which
'   can be accessed with random (direct) access.  File is physically sorted
'   so that a binary search technique can be used.  The maximum number of
'   checks that need to be made to determine if a meter is in the file and
'   find it is defined by the formula:
'   int(log(Records in file)/log(2))+1
'   This routine only reports to the operator and does not set any error or
'   warning flags.  It is up to the operator to make the decision if and how
'   to correct the record.
'============================================================================
SUB CheckRepairList (Value$(), Mfg%, Mbg%, Dfg%, Dbg%)
'----------------------------------------------------------------------------

    DIM RepData AS RepairData

'   Process serial number into Repairs.Asc format

    V$ = RIGHT$(STRING$(9, "0") + LTRIM$(RTRIM$(Value$(3))), 9)

'   Open the repair list file.

    RepDat% = FREEFILE
    OPEN "Repairs.asc" FOR RANDOM AS #RepDat% LEN = LEN(RepData) + 2

    IF LOF(RepDat%) < LEN(RepData) THEN
        CLOSE #RepDat%
        EXIT SUB
    END IF

'   Set the low and high limits

    Low% = 1

'   A repair list record, for these purposes, is 19 characters.  The extra two
'   that don't show in the field statement are the CR/LF pair put in the file
'   by Enable as it wrote the file as plain text.

    High% = LOF(RepDat%) \ 19 + 1

'   Now check the file to see if this serial number is in it.

    WHILE Serial$ <> V$ AND High% > Low% + 1
        Rec% = (High% + Low%) \ 2
        GET #RepDat%, Rec%, RepData

        Serial$ = RepData.Serial
        Code$ = RepData.Code
        Area$ = RepData.Area
        Mfr$ = RepData.Mfr

'       File serial number less than MRDES serial number

        IF Serial$ < V$ THEN
            Low% = Rec%
        END IF

'       File serial number greater than MRDES serial number

        IF Serial$ > V$ THEN
            High% = Rec%
        END IF
    WEND

'   Grab the information from the file, then close the file

    s$ = Serial$
    c$ = Code$
    a$ = Area$
    m$ = Mfr$
    t$ = Td$

    CLOSE #RepDat%

'   Default to no differences.

    Diff% = 0

'   Now see if we have a live one.

    IF s$ <> V$ THEN
        CLS
        COLOR Mfg%, Mbg%
        Cputs "Meter not found in STS file.", 10
        Cputs "Please check to see that you have the correct serial number", 12
        x$ = Get1Key$(CHR$(13), "Press ENTER to continue")
        EXIT SUB
    END IF

    CLS
    COLOR 15, 0
    Cputs "TRDES Check with STS Information", 1
    LOCATE 5, 30, 0
    PRINT "TRDES          STS"
    COLOR Mfg%, Mbg%
    PutXY 6, 10, "Serial Number"
    PutXY 6, 30, LTRIM$(RTRIM$(Value$(3)))
    s$ = LTRIM$(RTRIM$(s$))
    WHILE LEFT$(s$, 1) = "0" AND LEN(s$) > 1
        s$ = MID$(s$, 2)
    WEND

    PutXY 6, 45, s$
    PutXY 7, 10, "Code"
    PutXY 7, 30, Value$(1)
    PutXY 7, 45, c$

'   If there's a difference, identify it.

    IF c$ <> Value$(1) THEN
        Diff% = Diff% + 1
        COLOR Dfg%, Dbg%
        PutXY 7, 60, "Difference"
        COLOR Mfg%, Mbg%
    END IF

'   For the company, all we look at is the first character in the
'   Co/Div/Area.  It'll be a 0 or a 7

    PutXY 8, 10, "Company"

    IF LEFT$(Value$(10), 1) = "0" THEN
        Tcompany$ = "CSP"
    ELSE
        Tcompany$ = "OP"
    END IF
   
    PutXY 8, 30, Tcompany$
   
    IF a$ = "0" THEN
        Scompany$ = "CSP"
    ELSE
        Scompany$ = "OP"
    END IF
   
    PutXY 8, 45, Scompany$

    IF Tcompany$ <> Scompany$ THEN
        Diff% = Diff% + 1
        COLOR Dfg%, Dbg%
        PutXY 8, 60, "Difference"
        COLOR Mfg%, Mbg%
    END IF

    PutXY 9, 10, "Mfr"
    PutXY 9, 30, Value$(4)
    PutXY 9, 45, m$
    LOCATE 9, 60

    Mf$ = LTRIM$(RTRIM$(Value$(4)))

    IF m$ <> Mf$ THEN
        Diff% = Diff% + 1
        COLOR Dfg%, Dbg%
        PRINT "Difference"
        COLOR Mfg%, Mbg%
    END IF

'   One of these messages will tell the story.

    SELECT CASE Diff%
        CASE 0
            Cputs "No Differences found", 20
            COLOR Mfg%, Mbg%
            IF CkFlag% = 1 THEN
                EXIT SUB
            END IF
        CASE 1
            COLOR Dfg%, Dbg%
            Cputs "1 Difference found", 20
        CASE ELSE
            COLOR Dfg%, Dbg%
            Cputs STR$(Diff%) + " Differences found", 20
            COLOR Mfg%, Mbg%
    END SELECT

    x$ = Get1Key$(CHR$(13), "Press ENTER to continue")
'****************************************************************************
END SUB

SUB EditIniFile (FileName$, Company$, Facility$, Mfg%, Mbg%, Dfg%, Dbg%, CkFlag%)
'----------------------------------------------------------------------------
    DO
        COLOR 15, 0
        DrawMenuBox 1, 24, 4, 21, 78
        Cputs Company$, 2
        Cputs Facility$, 3
        PutXY 6, 20, "1 - Message character color" + STR$(Mfg%)
        PutXY 7, 20, "2 - Message background color" + STR$(Mbg%)
        PutXY 8, 20, "3 - Data field character color" + STR$(Dfg%)
        PutXY 9, 20, "4 - Data field background color" + STR$(Dbg%)

        Cputs Company$, 11

        Cputs Facility$, 13

        COLOR Mfg%, Mbg%
        PutXY 6, 60, "Character"
        COLOR Dfg%, Dbg%
        PutXY 8, 60, "Data field"

        Cputs "Select item to be changed by number 1 -4", 22
        Cputs "Press Enter to save changes and exit, or Esc to abandon changes and exit", 23

        DO
            PutXY 3, 3, DATE$
            PutXY 3, 69, TIME$
            PutXY 2, 3, DayOfWeek$(DATE$)

            CASE "1"
                LOCATE 17, 20
                INPUT ; "New message character color (0-15)", x%
                IF x% <> Mbg% AND x% > 0 AND x% < 16 THEN
                    Mfg% = x%
                END IF
            CASE "2"
                LOCATE 17, 20
                INPUT ; "New message background color (0-7):", x%
                IF x% <> Mfg% AND x% >= 0 AND x% < 8 THEN
                    Mbg% = x%
                END IF
            CASE "3"
                LOCATE 17, 20
                INPUT ; "New data field character color (0 - 15):", x%
                IF x% <> Dbg% AND x% >= 0 AND x% < 16 THEN
                    Dfg% = x%
                END IF
            CASE "4"
                LOCATE 17, 20
                INPUT ; "New data field background color (0-7):", x%
                IF x% <> Dfg% AND x% >= 0 AND x% < 8 THEN
                    Dbg% = x%
                END IF
            CASE "5"
                Cputs "New company header string (max 50 chars)", 17
                LOCATE 18, 20
                LINE INPUT x$
                IF x$ <> "" AND LEN(x$) < 50 THEN
                    Company$ = x$
                END IF
            CASE "6"
                Cputs "New facility header string (max 50 chars)", 17
                LOCATE 18, 20
                LINE INPUT x$
                IF x$ <> "" AND LEN(x$) < 50 THEN
                    Facility$ = x$
                END IF
            CASE "7"
                CkFlag% = ABS(CkFlag% = 0)
        END SELECT

        PRINT #IniFile%, Dfg%
        PRINT #IniFile%, Dbg%

    END IF

'****************************************************************************
'   Generate multiple records from template.
'============================================================================
SUB GenerateRecords (FileName$, Location%)
'----------------------------------------------------------------------------
   
    DIM Info AS InfoRecord, Buffer AS DataRecord
  
    CLS
    LOCATE 10, 20
    LINE INPUT "Number of transformer records to generate "; NumCards$
    NumCards% = VAL(NumCards$)

'   Allow the user to back out, if desired, by inputting zero

    IF NumCards% = 0 THEN
        CHAIN "MMENU"
    END IF

'   Open and check the file

    DataFile% = FREEFILE
    OPEN FileName$ FOR RANDOM AS #DataFile% LEN = 1000
    GET #DataFile%, 1, Info
    NumRec% = Info.Number

'   We accept a file with at least one record.

    IF NumRec% < 1 THEN
        CLOSE
        CLS
        PutXY 10, 27, "Not a valid template file!"
        PutXY 15, 17, "Press ENTER to return to Menu"
        x$ = Get1Key$(CHR$(13), "")
        EXIT SUB
    END IF

'   We have a valid template file.  Process.

    GET #1, NumRec% + 1, Buffer
   
    Work$ = Buffer.Buffer

'   Get strings to work with.

    Number$ = LTRIM$(RTRIM$(MID$(Work$, Location%, 26)))

'   This is the loop that does the deed.

    FOR NumRec% = NumRec% TO NumRec% + NumCards%
        PRINT Number$
        PUT #DataFile%, NumRec% + 1, Buffer
        Number$ = MID$(STR$(VAL(Number$) + 1), 2)
        MID$(Work$, Location%, 26) = Number$
        Buffer.Buffer = Work$
    NEXT

'   Put the number of records in the file in the pointer

    GET #DataFile%, 1, Info
    Info.Number = Info.Number + NumCards%
    Info.Signature = "TRDES"
    PUT #DataFile%, 1, Info

Split:
    CLOSE
'****************************************************************************
END SUB

'****************************************************************************
'   Round the quantity passed in Value$ to 2 places.  Make the returned
'   string Alen% characters long.
'============================================================================
FUNCTION Round$ (Value$, Alen%)
'----------------------------------------------------------------------------

'   If it's all spaces, there's nothing to do.

    IF Value$ = SPACE$(Alen%) THEN
        Round$ = Value$
        EXIT FUNCTION
    END IF

'   Something here.  Process.

    Value! = VAL(Value$)
    Value! = (INT((Value! * 100) + .5)) / 100
    V$ = LTRIM$(STR$(Value!))
    LengTh% = LEN(V$)
    DecPt% = INSTR(V$, ".")
    IF DecPt% = 0 THEN
        V$ = V$ + "."
    END IF
    DecPt% = INSTR(V$, ".")
    WHILE (LengTh% - DecPt%) < 2
        V$ = V$ + "0"
        LengTh% = LEN(V$)
    WEND
    Round$ = LEFT$(V$ + SPACE$(Alen%), Alen%)
'****************************************************************************
END FUNCTION
