SUB CT (Ptr%, NumRec%, Buffer$)

'----------------------------------------------------------------------------

	MID$(Template$, CompanyCode%, 12) = LTRIM$(RTRIM$(MID$(Buffer$, 901, 12)))

'   Mfr code

	MID$(Template$, MfrCode%, 2) = LTRIM$(RTRIM$(MID$(Buffer$, 28, 2)))

	MID$(Template$, DeviceCode%, 5) = MID$(Buffer$, 2, 5)
   
	Dcode$ = MID$(Template$, DeviceCode%, 5)

	MID$(Template$, DeviceSerialNumber%, 26) = LTRIM$(RTRIM$(MID$(Buffer$, 453, 26)))

'   Tester identification

	MID$(Template$, TesterIdentification%, 8) = MID$(Buffer$, 71, 3)

'   Test Ratio.  Possibly strange code allows dual ratio.

		Ratio$ = LEFT$(Ratio$, INSTR(Ratio$, ":") - 1)

	Slash% = INSTR(Ratio$, "/")

		LRatio$ = LEFT$(Ratio$, Slash% - 1)

'****************************************************************************
'   Prepares a record, then calls the record write routine.
'============================================================================
SUB DoaRecord (Template$, Buffer$, RatioLoc%, PhaseLoc%, AccClassLoc%)
'----------------------------------------------------------------------------

'   Ratio Accuracy
	  
	MID$(Template$, RatioAccuracy%, 5) = RightSet$(Left2$(MID$(Buffer$, RatioLoc%, 6)), 5, 4)

'   Phase Error

	Phase$ = LTRIM$(RTRIM$(MID$(Buffer$, PhaseLoc%, 6)))
	  
	MID$(Template$, PhaseErrorSign%, 1) = LEFT$(Phase$, 1)
   
	MID$(Template$, PhaseError%, 4) = RightSet$(MID$(Phase$, 2), 4, 2)

'   Accuracy Class

	MID$(Template$, AccuracyClass%, 3) = RightSet$(MID$(Buffer$, AccClassLoc%, 6), 3, 2)

'   If the primary to secondary/case volts is blank, default it.
   
	IF LTRIM$(RTRIM$(MID$(Template$, PrimaryToSecondaryCaseVolts%, 6))) = "" THEN
		MID$(Template$, PrimaryToSecondaryCaseVolts%, 6) = "000000"
	END IF

'   Write the record

	SELECT CASE MID$(Template$, CompanyCode%, 2)
		CASE "01"           ' Kingsport Power
			KgCount& = KgCount& + 1
		CASE "02"           ' Appalachian Power
			ApCount& = ApCount& + 1
		CASE "03"           ' Kentucky Power
			KtCount& = KtCount& + 1
		CASE "04"           ' Indiana Michigan Power
			IMCount& = IMCount& + 1
				CASE "05"           ' Indiana Michigan Power (Michigan)
			MiCount& = MiCount& + 1
		CASE "06"           ' Wheeling Power
			WpCount& = WpCount& + 1
		CASE "07"           ' Ohio Power
			OpCount& = OpCount& + 1
		CASE "10"           ' Columbus Southern Power
			CsCount& = CsCount& + 1
	END SELECT
   
	RTTemplate$ = RTRIM$(Template$)

	WriteRecord RTTemplate$, Buffer$

'****************************************************************************
END SUB

'****************************************************************************
'   This function effectively divides the value in Incoming$ by 100.
'   by locating the decimal point in the incoming string and moving
'   it two places to its left.  Since in this application the decimal point
'   will always be in the third or fourth position, there is very little
'   chance for a problem.
'============================================================================
FUNCTION Left2$ (Incoming$)
'----------------------------------------------------------------------------
   
	Work$ = LTRIM$(RTRIM$(Incoming$))
   
	Pointer% = INSTR(Work$, ".")
   
	IF Pointer% = 0 THEN
		Left2$ = Incoming$
	ELSE
		IF Pointer% = 3 THEN
			Left2$ = LEFT$("." + LEFT$(Work$, 2) + MID$(Work$, 4) + SPACE$(LEN(Incoming$)), LEN(Incoming$))
		ELSE
			Left2$ = LEFT$(LEFT$(Work$, 1) + "." + MID$(Work$, 2, 2) + MID$(Work$, 5) + SPACE$(LEN(Incoming$)), LEN(Incoming$))
		END IF
	END IF

'****************************************************************************
END FUNCTION

'****************************************************************************
'   Looks up primary/secondary and case Dielectric Test Volts.
'============================================================================
FUNCTION LookUpPriCaseVolts$ (Code$, TestType$)
'----------------------------------------------------------------------------
	SELECT CASE Code$
		CASE "0"                ' 600 volt
			Work$ = "002500"
		CASE "1"
			IF TestType$ = "1" THEN
				Work$ = "014000"
			ELSE
				Work$ = "019000"
			END IF
		CASE "3"                ' 15 KV
			IF TestType$ = "1" THEN
				Work$ = "025000"
			ELSE
				Work$ = "034000"
			END IF
		CASE "5"                ' 34.5 KV
			IF TestType$ = "1" THEN
				Work$ = "052500"
			ELSE
				Work$ = "070000"
			END IF
		CASE "7"                ' 69 KV
			IF TestType$ = "1" THEN
				Work$ = "105000"
			ELSE
				Work$ = "140000"
			END IF
		CASE "8"
			IF TestType$ = "1" THEN
				Work$ = "206000"
			ELSE
				Work$ = "275000"
			END IF
	END SELECT
	LookUpPriCaseVolts$ = Work$
'****************************************************************************
END FUNCTION

'****************************************************************************
'   Read menu.ini file.  File is a somewhat freeform text file.  The only
'   line actually required for this procedure to function is the line
'   beginning with an octothorpe, which is used to signal the end of the
'   file.
'============================================================================
SUB ReadIniFile (LocalFile$, Remote$, BcodeFile$, SerialFile$, Facility$, TestSet$)
'----------------------------------------------------------------------------
	CLS
	IniFile% = FREEFILE
	OPEN "menu.ini" FOR INPUT AS #IniFile%
	DO
		LINE INPUT #IniFile%, Work$
		Work$ = LTRIM$(RTRIM$(Work$))
		IF INSTR("#;", LEFT$(Work$, 1)) = 0 AND Work$ <> "" THEN
			IF UCASE$(LEFT$(Work$, 5)) = "LOCAL" THEN
				LocalFile$ = LTRIM$(MID$(Work$, INSTR(Work$, "=") + 1))
				IF INSTR(LocalFile$, " ") THEN
					LocalFile$ = LEFT$(LocalFile$, INSTR(LocalFile$, " ") - 1)
				END IF
			END IF
			IF UCASE$(LEFT$(Work$, 6)) = "REMOTE" THEN
				Remote$ = LTRIM$(MID$(Work$, INSTR(Work$, "=") + 1))
				IF INSTR(Remote$, " ") THEN
					Remote$ = LEFT$(Remote$, INSTR(LocalFile$, " ") - 1)
				END IF
			END IF
			IF UCASE$(LEFT$(Work$, 7)) = "BARCODE" THEN
				BcodeFile$ = LTRIM$(MID$(Work$, INSTR(Work$, "=") + 1))
				IF INSTR(BcodeFile$, " ") THEN
					Bcode$ = LEFT$(BcodeFile$, INSTR(BcodeFile$, " ") - 1)
				END IF
			END IF
			IF UCASE$(LEFT$(Work$, 6)) = "SERIAL" THEN
				SerialFile$ = LTRIM$(MID$(Work$, INSTR(Work$, "=") + 1))
				IF INSTR(SerialFile$, " ") THEN
					SerialFile$ = LEFT$(SerialFile$, INSTR(Serial$, " ") - 1)
				END IF
			END IF
			IF UCASE$(LEFT$(Work$, 8)) = "FACILITY" THEN
				Facility$ = LTRIM$(RTRIM$(MID$(Work$, INSTR(Work$, "=") + 1)))
			END IF
			IF UCASE$(LEFT$(Work$, 7)) = "TESTSET" THEN
				TestSet$ = LTRIM$(RTRIM$(MID$(Work$, INSTR(Work$, "=") + 1)))
			END IF
		END IF
	LOOP UNTIL LEFT$(Work$, 1) = "#"
	CLOSE #IniFile%
'****************************************************************************
END SUB

'****************************************************************************
'   Right sets a string representation of a value passed in Value$, either
'   integer or floating point, into a string of TotalChars% length, with
'   any decimal portion being Dec% characters in length.  Value$ should be
'   rounded to proper number of decmial places, if applicable, prior to
'   calling this function.  String returned will be all numbers, with leading
'   and trailing zeros as necessary to make the string TotalChar% in length.
'============================================================================
FUNCTION RightSet$ (Value$, TotalChar%, Dec%)
'----------------------------------------------------------------------------
	
	V$ = LTRIM$(RTRIM$(Value$))

'   Is there a decimal point?

	DecPt% = INSTR(V$, ".")
   
	IF DecPt% = 0 THEN
		V$ = V$ + "."
		DecPt% = INSTR(V$, ".")
	END IF

'   If there is, get the decimal part of the number into Dec$

	IF DecPt% > 0 AND DecPt% < LEN(V$) THEN
		Dec$ = MID$(V$, DecPt% + 1)
	END IF
 
'   Properly format the decimal portion

	IF Dec% > 0 THEN
		Dec$ = LEFT$(Dec$ + STRING$(Dec%, "0"), Dec%)
	END IF

'   If there's an integer portion, get it into Integer$

	IF DecPt% > 1 THEN
		Integer$ = LEFT$(V$, DecPt% - 1)
	END IF
  
'   Assemble and return the string

	RightSet$ = RIGHT$(STRING$(TotalChar%, "0") + Integer$ + Dec$, TotalChar%)
'****************************************************************************
END FUNCTION

'****************************************************************************
'   Opens the output file for append, writes the record.
'============================================================================
SUB WriteRecord (Work$, Buffer$)
'----------------------------------------------------------------------------

'   Numeric postions in interface record require numbers, not spaces.
'   Don't process header/trailer record

	IF INSTR("12", LEFT$(Work$, 1)) THEN
		MID$(Work$, KWHStopLoc%, KWHStopLen%) = "N" + STRING$(KWHStopLen% + 1, "0")
		MID$(Work$, KWHAccyLoc%, KWHAccyLen%) = STRING$(KWHAccyLen%, "0")
		MID$(Work$, TimingLoc%, TimingLen%) = STRING$(TimingLen%, "0")
		MID$(Work$, KVARHLoc%, KVARHLen%) = STRING$(KVARHLen%, "0")
		MID$(Work$, KVARHStopLoc%, KVARHStopLen%) = "N" + STRING$(KVARHStopLen% - 1, "0")
		MID$(Work$, MiscAccLoc%, MiscAccLen%) = STRING$(MiscAccLen%, "0")
		MID$(Work$, MiscStopLoc%, MiscStopLen%) = "N" + STRING$(MiscStopLen% - 1, "0")
		MID$(Work$, Ke1Loc%, Ke1Len%) = STRING$(Ke1Len%, "0")
		MID$(Work$, Ke2Loc%, Ke2Len%) = STRING$(Ke2Len%, "0")
		MID$(Work$, ResetLoc%, ResetLen%) = STRING$(ResetLen%, "0")
		MID$(Work$, TestSetLoc%, TestSetLen%) = TestSet$
		MID$(Work$, KWStopLoc%, KWStopLen%) = "N"
	END IF

'   MID$(Work$, 2, 12) = "07OH09009931"    we used to have everything from OP

	SELECT CASE RTRIM$(LTRIM$(MID$(Buffer$, 67, 4)))
		CASE "7009"
			ITSFROM$ = "07OH09009931"
		CASE "0007"
			ITSFROM$ = "07OH09009931"
		CASE "0004"
			ITSFROM$ = "04IN05610903"
		CASE ELSE
			ITSFROM$ = "07OH09009931"
	END SELECT

	MID$(Work$, 2, 12) = ITSFROM$

									
'   Now write the record

	Df% = FREEFILE
	OPEN "TRS2MDS" FOR APPEND AS #Df%
	PRINT #Df%, RTRIM$(Work$)
	CLOSE #Df%
'****************************************************************************
END SUB
