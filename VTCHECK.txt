'----------------------------------------------------------------------------

'----------------------------------------------------------------------------
	TYPE RepairData
		Serial AS STRING * 9        ' Device serial number
		Code AS STRING * 5          ' Device code
		Area AS STRING * 1          ' Actually this is now company
		Mfr AS STRING * 2           ' AEP Mfr code
		Ac AS STRING * 2
	END TYPE
'============================================================================
	TYPE InfoRecord
		Number AS INTEGER           ' Number of records in file
		Signature AS STRING * 20    ' TRDES Signature
	END TYPE

	TYPE DataRecord
		Buffer AS STRING * 1000     ' 1000 character record length
	END TYPE
'============================================================================
'   Declare procedures
'----------------------------------------------------------------------------
DECLARE FUNCTION AcCalc$ (Ratio$, phase$, Alen%, Sign%)
DECLARE FUNCTION AddSign$ (Work$, Alen%)
DECLARE FUNCTION AllCheck% (x$, Check$)
DECLARE FUNCTION CheckDate% (Value$)
DECLARE FUNCTION CheckDivision% (V$)
DECLARE FUNCTION CheckField% (Value$, Accept$)
DECLARE SUB CheckRatio (Value$, ClrFlag%, LineNumber$)
DECLARE FUNCTION CheckRegion% (V$)
DECLARE SUB CheckRepairList (Value$(), Mfg%, Mbg%, Dfg%, Dbg%)
DECLARE SUB Cputs (Message$, Row%)
DECLARE SUB ErrMsg (Msg$, ClrFlag%)
DECLARE FUNCTION Get1Key$ (Accept$, Prompt$)
DECLARE SUB MfrLookup (Value$, Vlen%, Eflag%, Estring$)
DECLARE FUNCTION OneDec% (Look$)
DECLARE SUB PrtScrn (Prow%(), Pcol%(), Prompt$(), Arow%(), Acol%(), Value$(), Count%, Company$, Facility$, Mfg%, Mbg%, Dfg%, Dbg%, Number%)
DECLARE SUB PutXY (Row%, Column%, Message$)

DECLARE FUNCTION Round$ (Value$, Alen%)
'============================================================================
'   $INCLUDE: 'VTDIMCOM.INC'
'============================================================================
'   File access records
'----------------------------------------------------------------------------
	DIM SHARED Info AS InfoRecord, DataRec AS DataRecord
'============================================================================
	Programmer$ = "K.W.Kroninger"
'   Version check

	IF COMMAND$ <> "" AND LEFT$(UCASE$(LTRIM$(COMMAND$)), 1) = "V" THEN
		PRINT "Voltage Transformer Input Data Check Utility - Version 9.00"
		END
	END IF
'============================================================================
'   Turn on error trap
'----------------------------------------------------------------------------
	ON ERROR GOTO NetProblem
'============================================================================
'   Set a couple of flags
   
	ClrFlag% = 0    ' Screen has not been cleared

'   Get the screen color right.

	COLOR Mfg%, Mbg%

'   Set up check strings

	Num$ = "0123456789"                     ' Numbers
	NumSp$ = Num$ + " "                     ' Numbers and space
	Alpha$ = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"   ' Letters
	NumPer$ = NumSp$ + " ."                 ' Numbers, space, period
	All$ = NumSp$ + Alpha$ + "-"            ' Numbers, letters, hyphen
'============================================================================
'   Setup is done.  Now do checks.
'----------------------------------------------------------------------------
'   Check code for all character positions filled with acceptable characters,
'   and the first character must be a "V".
   
	IF CheckField%(Value$(1), Alpha$ + Num$) = 1 OR LEFT$(Value$(1), 1) <> "V" THEN
		ErrMsg "Error - Code", ClrFlag%
	END IF

'   Activity Code field.  Format and check for acceptable values

	Value$(2) = RIGHT$(STRING$(Alen%(2), "0") + LTRIM$(RTRIM$(Value$(2))), Alen%(2))
	IF Value$(2) <> "01" AND Value$(2) <> "04" THEN
		ErrMsg "Error - Acvitivty Code", ClrFlag%
	END IF

'   Check Serial Number.  Remove leading zeros, check for valid characters.
   
	Value$(3) = LTRIM$(RTRIM$(Value$(3)))
   
	WHILE LEFT$(Value$(3), 1) = "0" AND LEN(Value$(3)) > 1
		Value$(3) = MID$(Value$(3), 2)
	WEND
   
	Value$(3) = LEFT$(Value$(3) + SPACE$(Alen%(3)), Alen%(3))
   
	IF AllCheck%(Value$(3), All$) = 1 OR LTRIM$(RTRIM$(Value$(3))) = "" THEN
		ErrMsg "Error - Serial Number", ClrFlag%
	END IF

'   Check Mfr with lookup table.
   
	MfrLookup Value$(4), Alen%(4), Flag%, " "

	IF AllCheck%(Value$(4), Alpha$ + Num$) = 1 OR Flag% = 1 THEN
		ErrMsg "Error - Manufacturer", ClrFlag%
	END IF

'   Check Volts.  Character check only.
   
	IF AllCheck%(Value$(6), All$ + "./") THEN
		ErrMsg "Error - Volts", ClrFlag%
	END IF

'   Check Dielectric Check.  Actually, just jam in a Y

	IF AllCheck%(Value$(9), " Y") THEN
		ErrMsg "Error - Delectric Check", ClrFlag%
	END IF

'   Check Region code against lookup table for compliance with MDS.

	Value$(10) = LEFT$(UCASE$(Value$(10)) + SPACE$(Alen%(10)), Alen%(10))
  
	IF CheckRegion%(Value$(10)) THEN
		ErrMsg "Error - Region", ClrFlag%
	END IF

'   Check Tester.  Check for valid characters and at least 3 characters.
   
	IF AllCheck%(Value$(11), Alpha$) = 1 OR LEN(LTRIM$(RTRIM$(Value$(11)))) < 3 THEN
		ErrMsg "Error - Tester ID", ClrFlag%
	END IF

'   Check Retirement Reason Code with table.
   
	RetireLookup Value$(12), Alen%(12), Eflag%, " "
   
	IF Eflag% = 1 THEN
		ErrMsg "Error - Retirment Reason Code", ClrFlag%
	END IF

'   Check Ratio.  Not a whole lot here, just check for characters.  As long
'   as this program's use is mainly by Greg and Ned, this simple check should
'   be enough.
   
	IF AllCheck%(Value$(15), NumPer$ + "/:") THEN
		ErrMsg "Error - Ratio", ClrFlag%
	END IF

'   Check Maximum Burden.  Check is just for acceptable characters.
   
	IF AllCheck%(Value$(16), All$ + ".") THEN
		ErrMsg "Error - Maximum Burden", ClrFlag%
	END IF

'   Volts line 1 - Check for valid characters and no more than one decimal
'   point.
   
	IF CheckField%(Value$(17), NumPer$) THEN
		ErrMsg "Error - Volts line 1", ClrFlag%
	END IF

'   Volts line 2  - Check for valid characters and no more than one decimal
'   point.
   
	IF CheckField%(Value$(18), NumPer$) THEN
		ErrMsg "Error - Volts line 2", ClrFlag%
	END IF

'   Volts line 3  - Check for valid characters and no more than one decimal
'   point.
   
	IF CheckField%(Value$(19), NumPer$) THEN
		ErrMsg "Error - Volts line 3", ClrFlag%
	END IF

'   Volts line 4  - Check for valid characters and no more than one decimal
'   point.
   
	IF CheckField%(Value$(20), NumPer$) THEN
		ErrMsg "Error - Volts line 4", ClrFlag%
	END IF

'   Volts line 5  - Check for valid characters and no more than one decimal
'   point.
   
	IF CheckField%(Value$(21), NumPer$) THEN
		ErrMsg "Error - Volts line 5", ClrFlag%
	END IF

'   Volts line 6 - Check for valid characters and no more than one decimal
'   point.
   
	IF CheckField%(Value$(22), NumPer$) THEN
		ErrMsg "Error - Volts line 6", ClrFlag%
	END IF

'   Burden line 1 - Check for valid characters and no more than one decimal
'   point.
   
	IF CheckField%(Value$(23), All$ + ".") THEN
		ErrMsg "Error - Burden line 1", ClrFlag%
	END IF

'   Burden line 2 - Check for valid characters and no more than one decimal
'   point.
   
	IF CheckField%(Value$(24), All$ + ".") THEN
		ErrMsg "Error - Burden line 2", ClrFlag%
	END IF

'   Burden line 3 - Check for valid characters and no more than one decimal
'   point.
   
	IF CheckField%(Value$(25), All$ + ".") THEN
		ErrMsg "Error - Burden line 3", ClrFlag%
	END IF

'   Burden line 4 - Check for valid characters and no more than one decimal
'   point.
   
	IF CheckField%(Value$(26), All$ + ".") THEN
		ErrMsg "Error - Burden line 4", ClrFlag%
	END IF

'   Burden line 5 - Check for valid characters and no more than one decimal
'   point.
   
	IF CheckField%(Value$(27), All$ + ".") THEN
		ErrMsg "Error - Burden line 5", ClrFlag%
	END IF

'   Burden line 6 - Check for valid characters and no more than one decimal
'   point.
   
	IF CheckField%(Value$(28), All$ + ".") THEN
		ErrMsg "Error - Burden line 6", ClrFlag%
	END IF

'   Ratio line 1.  Check for acceptable characters, no more than one decimal
'   point, round to two decimal places and check for accuracy within two
'   percent or specified by environment variable RATIO.

	IF CheckField%(Value$(30), NumPer$) THEN
		ErrMsg "Error - Ratio line 1", ClrFlag%
	END IF
   
	Value$(30) = Round$((Value$(30)), Alen%(30))

	CheckRatio Value$(30), ClrFlag%, "Line 1"

'   Phase Error line 1.  Check for acceptable characters and no more than one
'   decimal point, round to two decimal places, prefix sign character if
'   needed.
   
	IF CheckField(Value$(31), NumPer$ + "+-") THEN
		ErrMsg "Error - Phase Error line 1", ClrFlag%
	END IF
   
	Value$(31) = AddSign$(Round$((Value$(31)), Alen%(31)), Alen%(31))

'   Accuracy Class line 1.  Calculate the accuracy class, if necessary and
'   possible, round to two decimal places.
   
	IF Value$(32) = SPACE$(Alen%(32)) THEN
		IF Value$(30) <> SPACE$(Alen%(30)) THEN
			IF Value$(31) <> SPACE$(Alen%(31)) THEN
				Value$(32) = AcCalc$(Value$(30), Value$(31), Alen%(32), 1)
			END IF
		END IF
	END IF
   
	IF CheckField%(Value$(32), NumPer$) THEN
		ErrMsg "Error - Accuracy Class line 1", ClrFlag%
	END IF
   
	Value$(32) = Round$((Value$(32)), Alen%(32))

'   Ratio line 2.  Check for acceptable characters, no more than one decimal
'   point, round to two decimal places and check for accuracy within two
'   percent or specified by environment variable RATIO.
   
	IF CheckField%(Value$(34), NumPer$) THEN
		ErrMsg "Error - Ratio line 2", ClrFlag%
	END IF
   
	Value$(34) = Round$((Value$(34)), Alen%(34))

	CheckRatio Value$(34), ClrFlag%, "Line 2"

'   Phase Error line 2.  Check for acceptable characters and no more than one
'   decimal point, round to two decimal places, prefix sign character if
'   needed.
   
	IF CheckField%(Value$(35), NumPer$ + "+-") THEN
		ErrMsg "Error - Phase Error line 2", ClrFlag%
	END IF
   
	Value$(35) = AddSign$(Round$((Value$(35)), Alen%(35)), Alen%(35))

'   Accuracy Class line 2.  Calculate the accuracy class, if necessary and
'   possible, round to two decimal places.
   
	IF Value$(36) = SPACE$(Alen%(36)) THEN
		IF Value$(34) <> SPACE$(Alen%(34)) THEN
			IF Value$(35) <> SPACE$(Alen%(35)) THEN
				Value$(36) = AcCalc$(Value$(34), Value$(35), Alen%(36), 1)
			END IF
		END IF
	END IF
   
	IF CheckField%(Value$(36), NumPer$) THEN
		ErrMsg "Error - Accuracy Class line 2", ClrFlag%
	END IF
   
	Value$(36) = Round$((Value$(36)), Alen%(36))

'   Ratio line 3.  Check for acceptable characters, no more than one decimal
'   point, round to two decimal places and check for accuracy within two
'   percent or specified by environment variable RATIO.
   
	IF CheckField%(Value$(38), NumPer$) THEN
		ErrMsg "Error - Ratio line 3", ClrFlag%
	END IF
	Value$(38) = Round$((Value$(38)), Alen%(38))

	CheckRatio Value$(38), ClrFlag%, "Line 3"

'   Phase Error line 3.  Check for acceptable characters and no more than one
'   decimal point, round to two decimal places, prefix sign character if
'   needed.
   
	IF CheckField%(Value$(39), NumPer$ + "+-") THEN
		ErrMsg "Error - Phase Error line 3", ClrFlag%
	END IF
   
	Value$(39) = AddSign$(Round$((Value$(39)), Alen%(39)), Alen%(39))

'   Accuracy Class line 3.  Calculate the accuracy class, if necessary and
'   possible, round to two decimal places.
   
	IF Value$(40) = SPACE$(Alen%(40)) THEN
		IF Value$(38) <> SPACE$(Alen%(38)) THEN
			IF Value$(39) <> SPACE$(Alen%(39)) THEN
				Value$(40) = AcCalc$(Value$(38), Value$(39), Alen%(40), 1)
			END IF
		END IF
	END IF
   
	IF CheckField%(Value$(40), NumPer$) THEN
		ErrMsg "Error - Accuracy Class line 3", ClrFlag%
	END IF
   
	Value$(40) = Round$((Value$(40)), Alen%(40))

'   Ratio line 4.  Check for acceptable characters, no more than one decimal
'   point, round to two decimal places and check for accuracy within two
'   percent or specified by environment variable RATIO.
   
	IF CheckField%(Value$(42), NumPer$) THEN
		ErrMsg "Error - Ratio line 4", ClrFlag%
	END IF
   
	Value$(42) = Round$((Value$(42)), Alen%(42))

	CheckRatio Value$(42), ClrFlag%, "Line 4"

'   Phase Error line 4.  Check for acceptable characters and no more than one
'   decimal point, round to two decimal places, prefix sign character if
'   needed.
   
	IF CheckField%(Value$(43), NumPer$ + "+-") THEN
		ErrMsg "Error - Phase Error line 4", ClrFlag%
	END IF
	Value$(43) = AddSign$(Round$((Value$(43)), Alen%(43)), Alen%(43))

'   Accuracy Class line 4.  Calculate the accuracy class, if necessary and
'   possible, round to two decimal places.
   
	IF Value$(44) = SPACE$(Alen%(44)) THEN
		IF Value$(42) <> SPACE$(Alen%(42)) THEN
			IF Value$(43) <> SPACE$(Alen%(43)) THEN
				Value$(44) = AcCalc$(Value$(42), Value$(43), Alen%(44), 1)
			END IF
		END IF
	END IF
   
	IF CheckField%(Value$(44), NumPer$) THEN
		ErrMsg "Error - Accuracy Class line 4", ClrFlag%
	END IF
   
	Value$(44) = Round$((Value$(44)), Alen%(44))

'   Ratio line 5.  Check for acceptable characters, no more than one decimal
'   point, round to two decimal places and check for accuracy within two
'   percent or specified by environment variable RATIO.
   
	IF CheckField%(Value$(46), NumPer$) THEN
		ErrMsg "Error - Ratio line 5", ClrFlag%
	END IF
   
	Value$(46) = Round$((Value$(46)), Alen%(46))

	CheckRatio Value$(46), ClrFlag%, "Line 5"

'   Phase Error line 5.  Check for acceptable characters and no more than one
'   decimal point, round to two decimal places.
   
	IF CheckField%(Value$(47), NumPer$ + "+-") THEN
		ErrMsg "Error - Phase Error line 5", ClrFlag%
	END IF
	Value$(47) = AddSign$(Round$((Value$(47)), Alen%(47)), Alen%(47))

'   Accuracy Class line 5.  Calculate the accuracy class, if necessary and
'   possible, round to two decimal places.
   
	IF Value$(48) = SPACE$(Alen%(48)) THEN
		IF Value$(46) <> SPACE$(Alen%(46)) THEN
			IF Value$(47) <> SPACE$(Alen%(47)) THEN
				Value$(48) = AcCalc$(Value$(46), Value$(47), Alen%(48), 1)
			END IF
		END IF
	END IF
   
	IF CheckField%(Value$(48), NumPer$) THEN
		ErrMsg "Error - Accuracy Class line 5", ClrFlag%
	END IF
   
	Value$(48) = Round$((Value$(48)), Alen%(48))

'   Ratio line 6.  Check for acceptable characters, no more than one decimal
'   point, round to two decimal places and check for accuracy within two
'   percent or specified by environment variable RATIO.
   
	IF CheckField%(Value$(50), NumPer$) THEN
		ErrMsg "Error - Ratio line 6", ClrFlag%
	END IF
   
	Value$(50) = Round$((Value$(50)), Alen%(50))

	CheckRatio Value$(50), ClrFlag%, "Line 6"

'   Phase Error line 6.  Check for acceptable characters and no more than one
'   decimal point, round to two decimal places.
   
	IF CheckField%(Value$(51), NumPer$ + "+-") THEN
		ErrMsg "Error - Phase Error line 6", ClrFlag%
	END IF
   
	Value$(51) = AddSign$(Round$((Value$(51)), Alen%(51)), Alen%(51))

'   Accuracy Class line 6.  Calculate the accuracy class, if necessary and
'   possible, round to two decimal places.
   
	IF Value$(52) = SPACE$(Alen%(52)) THEN
		IF Value$(50) <> SPACE$(Alen%(50)) THEN
			IF Value$(51) <> SPACE$(Alen%(51)) THEN
				Value$(52) = AcCalc$(Value$(50), Value$(51), Alen%(52), 1)
			END IF
		END IF
	END IF
   
	IF CheckField%(Value$(52), NumPer$) THEN
		ErrMsg "Error - Accuracy Class line 6", ClrFlag%
	END IF
   
	Value$(52) = Round$((Value$(52)), Alen%(52))

'   Check Test Date
   
	IF CheckDate%(Value$(53)) THEN
		ErrMsg "Error - Test Date", ClrFlag%
	END IF

'   Check Price
   
	IF AllCheck%(Value$(54), NumPer$ + "$") THEN
		ErrMsg "Error - Price", ClrFlag%
	END IF

'   Check damage and defect codes

	FOR x% = 55 TO 60
		IF Value$(x%) <> SPACE$(Alen%(x%)) THEN
			Value$(x%) = RIGHT$("00" + LTRIM$(RTRIM$(Value$(x%))), Alen%(x%))
			IF AllCheck%(Value$(x%), Num$) THEN
				IF x% < 58 THEN
					ErrMsg "Error - Damage code", ClrFlag%
				ELSE
					ErrMsg "Error - Defect code", ClrFlag%
				END IF
			END IF
		END IF
	NEXT

'   Check for same damage/defect code entered more then once

	IF Value$(55) = "  " AND Value$(56) <> "  " THEN
		SWAP Value$(55), Value$(56)
	END IF
   
	IF Value$(56) = "  " AND Value$(57) <> "  " THEN
		SWAP Value$(56), Value$(57)
		IF Value$(55) = "  " AND Value$(56) <> "  " THEN
			SWAP Value$(55), Value$(56)
		END IF
	END IF
   
	IF Value$(56) <> "  " AND Value$(55) = Value$(56) THEN
		ErrMsg "Duplicate damage codes", ClrFlag%
	END IF
   
	IF Value$(57) <> "  " AND (Value$(55) = Value$(57) OR Value$(56) = Value$(57)) THEN
		ErrMsg "Duplicate damage codes", ClrFlag%
	END IF
		
	IF Value$(58) = "  " AND Value$(59) <> "  " THEN
		SWAP Value$(58), Value$(59)
	END IF
   
	IF Value$(59) = "  " AND Value$(60) <> "  " THEN
		SWAP Value$(59), Value$(60)
		IF Value$(58) = "  " AND Value$(59) <> "  " THEN
			SWAP Value$(58), Value$(59)
		END IF
	END IF
   
	IF Value$(59) <> "  " AND Value$(58) = Value$(59) THEN
		ErrMsg "Duplicate defect codes", ClrFlag%
	END IF
   
	IF Value$(60) <> "  " AND (Value$(58) = Value$(60) OR Value$(59) = Value$(60)) THEN
		ErrMsg "Duplicate defect codes", ClrFlag%
	END IF

'   Check against STS information, if it's available

	IF LTRIM$(RTRIM$(Value$(2))) = "" AND ClrFlag% = 0 THEN
		CheckRepairList Value$(), Mfg%, Mbg%, Dfg%, Dbg%
	END IF

'============================================================================
'   Done with checks. Write it?
'----------------------------------------------------------------------------
'   It's bad.  Inform user, then return to data entry/edit

	IF ClrFlag% = 1 THEN
		x$ = Get1Key$(CHR$(13), "Press ENTER to return to edit screen")
		CallPrg% = 2
		CHAIN "Vtdata"
	END IF
'----------------------------------------------------------------------------
'   If it's check only, just return
   
Split:
	IF Flag% = 1 THEN
		CallPrg% = 2
		CHAIN "Vtdata"
	END IF
'============================================================================
'   All checks passed.  Write it?
   
	PrtScrn Prow%(), Pcol%(), Prompt$(), Arow%(), Acol%(), Value$(), 60, Company$, Facility$, Mfg%, Mbg%, Dfg%, Dbg%, 60
   
	COLOR Mfg%, Mbg%
   
	x$ = Get1Key$("YN", "Data checks OK.  Write it? (Y/N)")
   
	IF x$ = "N" THEN
		CallPrg% = 2
		CHAIN "Vtdata"
	END IF
'============================================================================
'   Put the data into the buffer, and write the record
'============================================================================
'   Establish string

	Buffer$ = SPACE$(1000)
'============================================================================
'   ID record type
'   Record types are as follows:
'       1=Current Transformer
'       3=Potential Transformer
'       5=Phase Shift Transformer
'       7=Metering Outfit

	MID$(Buffer$, 1, 1) = "3"
'============================================================================
'   Put the data into the buffer
'----------------------------------------------------------------------------
	MID$(Buffer$, 2, Alen%(1)) = Value$(1)      '   Device Code
	MID$(Buffer$, 991, Alen%(2)) = Value$(2)    '   Activity Code
	MID$(Buffer$, 453, Alen%(3)) = Value$(3)    '   MFG Serial Number
	MID$(Buffer$, 28, Alen%(4)) = Value$(4)     '   Mfg
	MID$(Buffer$, 33, Alen%(5)) = Value$(5)     '   Type
	MID$(Buffer$, 481, Alen%(6)) = Value$(6)    '   Volts
	MID$(Buffer$, 66, Alen%(9)) = Value$(9)     '   Dielectric Check
	MID$(Buffer$, 901, Alen%(10)) = Value$(10)  '   Division
	MID$(Buffer$, 71, Alen%(11)) = Value$(11)   '   Tester
	MID$(Buffer$, 479, Alen%(12)) = Value$(12)  '   Retirement Code
	MID$(Buffer$, 75, Alen%(13)) = Value$(13)   '   Remarks
	MID$(Buffer$, 153, Alen%(15)) = Value$(15)  '   Serial number
	MID$(Buffer$, 162, Alen%(16)) = Value$(16)  '   Maximum Burden
	MID$(Buffer$, 165, Alen%(17)) = Value$(17)  '   Volts line 1
	MID$(Buffer$, 170, Alen%(18)) = Value$(18)  '   Volts line 2
	MID$(Buffer$, 175, Alen%(19)) = Value$(19)  '   Volts line 3
	MID$(Buffer$, 195, Alen%(20)) = Value$(20)  '   Volts line 4
	MID$(Buffer$, 200, Alen%(21)) = Value$(21)  '   Volts line 5
	MID$(Buffer$, 205, Alen%(22)) = Value$(22)  '   Volts line 6
	MID$(Buffer$, 210, Alen%(23)) = Value$(23)  '   Burden line 1
	MID$(Buffer$, 213, Alen%(24)) = Value$(24)  '   Burden line 2
	MID$(Buffer$, 216, Alen%(25)) = Value$(25)  '   Burden line 3
	MID$(Buffer$, 228, Alen%(26)) = Value$(26)  '   Burden line 4
	MID$(Buffer$, 231, Alen%(27)) = Value$(27)  '   Burden line 5
	MID$(Buffer$, 234, Alen%(28)) = Value$(28)  '   Burden line 6
	MID$(Buffer$, 243, Alen%(30)) = Value$(30)  '   Ratio line 1
	MID$(Buffer$, 249, Alen%(31)) = Value$(31)  '   Phase Error line 1
	MID$(Buffer$, 255, Alen%(32)) = Value$(32)  '   Accuracy Class line 1
	MID$(Buffer$, 267, Alen%(34)) = Value$(34)  '   Ratio line 2
	MID$(Buffer$, 273, Alen%(35)) = Value$(35)  '   Phase Error line 2
	MID$(Buffer$, 279, Alen%(36)) = Value$(36)  '   Accuracy Class line 2
	MID$(Buffer$, 291, Alen%(38)) = Value$(38)  '   Ratio line 3
	MID$(Buffer$, 297, Alen%(39)) = Value$(39)  '   Phase Error line 3
	MID$(Buffer$, 303, Alen%(40)) = Value$(40)  '   Accuracy Class line 3
	MID$(Buffer$, 387, Alen%(42)) = Value$(42)  '   Ratio line 4
	MID$(Buffer$, 393, Alen%(43)) = Value$(43)  '   Phase Error line 4
	MID$(Buffer$, 399, Alen%(44)) = Value$(44)  '   Accuracy Class line 4
	MID$(Buffer$, 411, Alen%(46)) = Value$(46)  '   Ratio line 5
	MID$(Buffer$, 417, Alen%(47)) = Value$(47)  '   Phase Error line 5
	MID$(Buffer$, 423, Alen%(48)) = Value$(48)  '   Accuracy Class line 5
	MID$(Buffer$, 435, Alen%(50)) = Value$(50)  '   Ratio line 6
	MID$(Buffer$, 441, Alen%(51)) = Value$(51)  '   Phase Error line 6
	MID$(Buffer$, 447, Alen%(52)) = Value$(52)  '   Accuracy Class line 6
	MID$(Buffer$, 125, Alen%(53)) = Value$(53)  '   Test Date
	MID$(Buffer$, 133, Alen%(54)) = Value$(54)  '   Price
	MID$(Buffer$, 601, Alen%(55)) = Value$(55)  '   Damage code 1
	MID$(Buffer$, 603, Alen%(56)) = Value$(56)  '   Damage code 2
	MID$(Buffer$, 605, Alen%(57)) = Value$(57)  '   Damage code 3
	MID$(Buffer$, 607, Alen%(58)) = Value$(58)  '   Defect code 1
	MID$(Buffer$, 609, Alen%(59)) = Value$(59)  '   Defect code 2
	MID$(Buffer$, 611, Alen%(60)) = Value$(60)  '   Defect code 3
	MID$(Buffer$, 993, 8) = TIME$               '   Time Stamp
'----------------------------------------------------------------------------
'   Write the record to the disk file.

	DataFile% = FREEFILE
	OPEN FileName$ FOR RANDOM AS #DataFile% LEN = LEN(DataRec)
	GET #DataFile%, 1, Info
	RecNum% = Info.Number
	DataRec.Buffer = Buffer$

'   If we're editing a record, put it back in place.  If it's a new
'   record, add it to the end of the file.

	IF EdRec% > 0 THEN
		RecNum% = EdRec%
	ELSE
		RecNum% = RecNum% + 1
	END IF
	PUT #DataFile%, RecNum% + 1, DataRec
	IF EdRec% = 0 THEN
		Info.Number = RecNum%
		Info.Signature = "TRDES"
		PUT #DataFile%, 1, Info
	END IF
