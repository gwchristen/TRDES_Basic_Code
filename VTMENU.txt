	NEXT

			CASE "4"

			Buff$ = DataRec.Buffer
			Typ$ = LEFT$(Buff$, 1)
			IF Typ$ = "3" THEN
				Parse Buff$
				PrtScrn Prow%(), Pcol%(), Prompt$(), Arow%(), Acol%(), Value$(), 60, Company$, Facility$, Mfg%, Mbg%, Dfg%, Dbg%, 60

			END IF
		END IF
	NEXT

	PutXY 25, 1, SPACE$(79)
	GOTO Menu
'============================================================================
'   List records on file
'----------------------------------------------------------------------------
RecsOnFile:
	CLS
	COLOR Mfg%, Mbg%
	PutXY 25, 1, SPACE$(79)
	GOSUB OpnFile
	IF RecNum% = 0 THEN
		GOTO Menu
	END IF
	LineCount% = Heading%
   
'   Clear the escape flag

	x$ = " "
	FOR Ptr% = 1 TO RecNum%
		IF x$ <> CHR$(27) THEN
			GET #1, Ptr% + 1, DataRec
			Buff$ = DataRec.Buffer
			x$ = LEFT$(Buff$, 1)
			IF x$ = "3" THEN
				Parse Buff$
				LineCount% = PrintLine%(LineCount%, Ptr%, Value$())
			END IF
			IF LineCount% = 20 THEN
				x$ = Get1Key$(CHR$(13) + CHR$(27), "ENTER to continue, ESC to return to menu")
				LineCount% = 0
				IF x$ = CHR$(13) THEN
					LineCount% = PrintHeading%
				END IF
			END IF
		END IF
	NEXT
	CLOSE
	IF LineCount% > 0 THEN
		x$ = Get1Key$(CHR$(13), "Press ENTER to return to the menu")
	END IF
	GOTO Menu
'============================================================================
'   Open file subroutine
'----------------------------------------------------------------------------
OpnFile:
	CLS
	COLOR Mfg%, Mbg%
	PutXY 25, 1, SPACE$(79)
	OPEN FileName$ FOR RANDOM AS #1 LEN = LEN(DataRec)
	GET #1, 1, Info
	RecNum% = Info.Number
	IF RecNum% = 0 THEN
		CLOSE
		KILL FileName$
		CLS
		PutXY 10, 20, "No file found on this disk."
		PutXY 15, 20, "Try again? (Y/N)"
		x$ = Get1Key$("YN", "")
		IF x$ = "N" THEN
			RETURN
		END IF
		CLS
		PutXY 25, 23, "Insert new disk, then press ENTER"
		x$ = Get1Key$(CHR$(13), "")

		GOTO OpnFile
	END IF
   
	RETURN
'============================================================================
'   Subroutine to edit record
'----------------------------------------------------------------------------
EditRecord:
	CLS
	GOSUB OpnFile
	CLS
	IF RecNum% = 0 THEN
		GOTO Menu
	END IF
	LOCATE
	PRINT USING "Record number to edit (1-####)"; RecNum%
	INPUT EdRec%

'   If the user wishes to back out, an entry of zero or
'   more records than there are in the file will do so.

	IF EdRec% < 1 OR EdRec% > RecNum% THEN
		CLOSE
		GOTO Menu
	END IF
	GET #1, EdRec% + 1, DataRec
	Buff$ = DataRec.Buffer
	Parse Buff$
	x$ = LEFT$(Buff$, 1)
	IF x$ <> "3" THEN
		CLOSE
		GOTO Menu
	END IF
	CallPrg% = 0
	Flag% = 4
	CLOSE
	CHAIN "VTDATA"
'============================================================================
'   Error traps
'----------------------------------------------------------------------------
BadDrive:
	PutXY 10, 18, "There seems to be a problem with the drive."
	PutXY 15, 24, "Please correct it and press ENTER"
	x$ = Get1Key$(CHR$(13), "")
	RESUME OpenIt
'----------------------------------------------------------------------------
'   Network Problems

NetProblem:
   
	IF UCASE$(ENVIRON$("DEBUG")) = "YES" THEN
	   
		Junk$ = Get1Key$(CHR$(13) + CHR$(27), STR$(ERR))
   
	ELSE
	  
'   $INCLUDE: 'c:\mrswork\neterror.inc'

	END IF
   
	IF Junk$ = CHR$(27) THEN
		RESUME Menu
	ELSE
		RESUME
	END IF

'   End of error traps
'****************************************************************************
'   End of VTMenu main module

'****************************************************************************
'   Clears screen, prints menu.  Returns menu choice.
'============================================================================
FUNCTION PrintMenu$
'----------------------------------------------------------------------------

	SHARED FileName$

	CLS
 
	COLOR Mfg%, Mbg%

	InFile% = FREEFILE
	OPEN FileName$ FOR RANDOM AS #InFile% LEN = LEN(Buffer)
	GET #InFile%, 1, Info
	CLOSE #InFile%

'   Clear Status Line

	LOCATE 25, 1, 0
	PRINT SPACE$(79);

'   Locate and draw a box to define menu
	 
	DrawMenuBox 1, 24, 6, 20, 78

'   Print Menu Text

	Cputs Company$, 2
	Cputs Facility$, 3
	Cputs "Transformer Record Data Entry System", 4
	Cputs "Voltage Transformer Menu", 5

'   Menu choice text

	PutXY 8, 24, "1. List Transformer Test Records"
	PutXY 9, 24, "2. List of Transformers in File"
	PutXY 10, 24, "3. Enter Voltage Transformer data"
	PutXY 11, 24, "4. Edit Voltage Transformer Data"
	PutXY 12, 24, "5. Generate VT Records"
	PutXY 14, 24, "6. Exit"

	PutXY 16, 24, "C - Edit colors"

	Cputs STR$(Info.Number) + " records in file.", 18
   
	PutXY 22, 32, "Option number? "

'   Get input
 
	PutXY 2, 77 - LEN(Version$), Version$
 
	COLOR Mfg%, Mbg%
	DO
		DO
			PutXY 5, 3, DATE$
			PutXY 5, 69, TIME$
			PutXY 2, 3, DayOfWeek$(DATE$)
			LOCATE , , 0
			x$ = UCASE$(INKEY$)
		LOOP WHILE x$ = ""
		IF x$ = CHR$(27) THEN
			x$ = "6"
		END IF
	LOOP WHILE INSTR("123456C", x$) = 0

	PrintMenu$ = x$
'****************************************************************************
END FUNCTION
